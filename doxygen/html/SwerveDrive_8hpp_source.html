268 frc::ChassisSpeeds DriveToPoseSetpoint(frc::Pose2d targetPose, frc::Pose2d currentPose,
269 frc::Rotation2d heading) {
270 auto rotationPIDRef = m_config->GetRotationPID();
271 if (!rotationPIDRef) {
272 throw exceptions::SwerveDriveConfigurationException(
273 "No rotation PID controller configured, cannot drive to pose. Use "
274 "SwerveDriveConfig::WithRotationController(PIDController) to fix this error.");
275 }
276 auto translationPIDRef = m_config->GetTranslationPID();
277 if (!translationPIDRef) {
278 throw exceptions::SwerveDriveConfigurationException(
279 "No translation PID controller configured, cannot drive to pose. Use "
280 "SwerveDriveConfig::WithTranslationController(PIDController) to fix this error.");
281 }
282 auto& rotationPID = rotationPIDRef->get();
283 auto& translationPID = translationPIDRef->get();
284 // Plain field-frame translation delta (not RelativeTo/operator-, which expresses the result
285 // in targetPose's rotated frame and would skew the commanded direction whenever targetPose's
286 // heading is non-zero).
287 auto toTarget = targetPose.Translation() - currentPose.Translation();
288 double distance = toTarget.Norm().value();
289 // The controller sees the distance as its measurement with a setpoint of 0, so its output is
290 // negative while there is distance left: the speed toward the target is the negative of it.
291 double speed = -translationPID.Calculate(distance, 0.0);
292 // Drive at that speed along the unit vector toward the target.
293 double vx = distance > 1e-9 ? toTarget.X().value() / distance * speed : 0.0;
294 double vy = distance > 1e-9 ? toTarget.Y().value() / distance * speed : 0.0;
295 return frc::ChassisSpeeds::FromFieldRelativeSpeeds(
296 frc::ChassisSpeeds{
297 units::meters_per_second_t{vx}, units::meters_per_second_t{vy},
298 units::radians_per_second_t{
299 rotationPID.Calculate(currentPose.Rotation().Radians().value(),
300 targetPose.Rotation().Radians().value())}},
301 heading);
302 }

437 void UpdateTelemetry() {
438 UpdatePoseEstimator();
439 auto pose = GetPose();
440 auto currentStates = GetModuleStates();
441
442 m_swerveTelemetry->Publish(this);
443
444 for (auto* mod : m_config->GetModules()) {
445 mod->UpdateTelemetry();
446 }
447 m_telemetry.UpdateLoopTime();
448
449 m_field2d.SetRobotPose(pose);
450 std::vector<frc::Pose2d> modulePoses;
451 modulePoses.reserve(NumModules);
452 for (size_t i = 0; i < NumModules; ++i) {
453 auto* mod = m_config->GetModules()[i];
454 auto location = *mod->GetConfig().GetLocation();
455 auto moduleTranslation = pose.Translation() + location.RotateBy(pose.Rotation());
456 auto moduleHeading = pose.Rotation() + currentStates[i].angle;
457 modulePoses.push_back(frc::Pose2d{moduleTranslation, moduleHeading});
458 }
459 m_field2d.GetObject("modules")->SetPoses(modulePoses);
460 }

469 void SimIterate() {
470 if (!m_simTimer.IsRunning()) m_simTimer.Start();
471 for (auto* mod : m_config->GetModules()) {
472 mod->SimIterate();
473 }
474 auto desired = m_kinematics.ToChassisSpeeds(m_desiredModuleStates);
475
476 auto dt = m_simTimer.Get();
477 frc::Twist2d twist{desired.vx * dt, desired.vy * dt, desired.omega * dt};
478 m_simPose = m_simPose.Exp(twist);
479
480 auto speeds = m_kinematics.ToChassisSpeeds(GetModuleStates());
481 m_simGyroAngle +=
482 units::degree_t{units::degrees_per_second_t{speeds.omega}.value() * dt.value()};
483 m_simTimer.Reset();
484 }

678 void SetupTelemetry() {
679 using TelemetryVerbosity = motorcontrollers::SmartMotorControllerConfig::TelemetryVerbosity;
680 telemetry::SwerveDriveTelemetryConfig telemetryCfg;
681 if (auto specified = m_config->GetSwerveDriveTelemetryConfig()) {
682 telemetryCfg = std::move(*specified);
683 } else {
684 telemetryCfg.WithTelemetryVerbosity(
685 m_config->GetTelemetryVerbosity().value_or(TelemetryVerbosity::HIGH));
686 }
687
688 if (auto dataLogName = telemetryCfg.GetDataLogName()) {
689 m_telemetry.SetupTelemetry(GetName(), *dataLogName);
690 } else {
691 m_telemetry.SetupTelemetry(GetName());
692 }
693
694 m_swerveTelemetry.emplace(std::move(telemetryCfg));
695 m_swerveTelemetry->SetupTelemetry(this);
696
697 // Drive to pose tuning needs both controllers, so only offer it when they are configured.
698 if (m_config->GetTranslationPID() && m_config->GetRotationPID()) {
699 m_driveToPoseTuningCommand.emplace(frc2::cmd::StartRun(
700 [this] {
701 std::puts(
702 "================= Starting SwerveDrive.driveToPoseTuning() =================\n");
703 ResetTranslationPID();
704 ResetRotationPID();
705 },
706 [this] { m_swerveTelemetry->ApplyTuningValues(this); }));
707 frc::SmartDashboard::PutData("Mechanisms/" + GetName() + "/tuning/driveToPose",
708 m_driveToPoseTuningCommand->get());
709 }
710 }

758// SwerveDriveTelemetry::SetupTelemetry()/Publish()/ApplyTuningValues() are templated on the
759// SwerveDrive's module count so that SwerveDriveTelemetry itself need not be a template. Their
760// bodies live here — after SwerveDrive<NumModules> is fully defined above — because this is the
761// only place both SwerveDriveTelemetry's private members and SwerveDrive<NumModules>'s interface
762// are simultaneously visible.
763
764template <std::size_t NumModules>
765void SwerveDriveTelemetry::SetupTelemetry(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
766 auto driveName = drive->GetName();
767 for (auto* module : drive->GetConfig().GetModules()) {
768 module->SetupTelemetry(driveName);
769 }
770
771 auto inst = nt::NetworkTableInstance::GetDefault();
772 m_dataTable = inst.GetTable("Mechanisms")->GetSubTable(driveName);
773 m_tuningTable = inst.GetTable("Tuning")->GetSubTable(driveName);
774
775 bool nt4Enabled = m_config.GetNT4Enabled();
776 auto dataLogName = m_config.GetDataLogName();
777
778 {
779 // Only publish the drive to pose gains and auto-align fields for controllers that exist.
780 auto translationPID = drive->GetConfig().GetTranslationPID();
781 auto rotationPID = drive->GetConfig().GetRotationPID();
782 auto& doubleFields = m_config.GetDoubleFields();
783 if (translationPID) {
784 doubleFields.at(DoubleTelemetryField::TranslationP)
785 .SetDefaultValue(translationPID->get().GetP());
786 doubleFields.at(DoubleTelemetryField::TranslationI)
787 .SetDefaultValue(translationPID->get().GetI());
788 doubleFields.at(DoubleTelemetryField::TranslationD)
789 .SetDefaultValue(translationPID->get().GetD());
790 } else {
791 doubleFields.at(DoubleTelemetryField::TranslationP).Disable();
792 doubleFields.at(DoubleTelemetryField::TranslationI).Disable();
793 doubleFields.at(DoubleTelemetryField::TranslationD).Disable();
794 }
795 if (rotationPID) {
796 doubleFields.at(DoubleTelemetryField::RotationP).SetDefaultValue(rotationPID->get().GetP());
797 doubleFields.at(DoubleTelemetryField::RotationI).SetDefaultValue(rotationPID->get().GetI());
798 doubleFields.at(DoubleTelemetryField::RotationD).SetDefaultValue(rotationPID->get().GetD());
799 } else {
800 doubleFields.at(DoubleTelemetryField::RotationP).Disable();
801 doubleFields.at(DoubleTelemetryField::RotationI).Disable();
802 doubleFields.at(DoubleTelemetryField::RotationD).Disable();
803 }
804 // Auto-align drives to a pose, which needs both controllers.
805 if (!translationPID || !rotationPID) {
806 doubleFields.at(DoubleTelemetryField::AutoAlignPoseX).Disable();
807 doubleFields.at(DoubleTelemetryField::AutoAlignPoseY).Disable();
808 doubleFields.at(DoubleTelemetryField::AutoAlignPoseRotation).Disable();
809 m_config.GetBoolFields().at(BooleanTelemetryField::AutoAlignEnabled).Disable();
810 }
811
812 auto& modules = drive->GetConfig().GetModules();
813 if (!modules.empty()) {
814 auto* driveMotor = modules[0]->GetDriveMotorController();
815 auto driveGains =
816 driveMotor->GetConfig().GetSlotGains(driveMotor->GetClosedLoopControllerSlot());
817 m_config.GetDoubleFields()
818 .at(DoubleTelemetryField::ModulesDriveP)
819 .SetDefaultValue(driveGains.kP);
820 m_config.GetDoubleFields()
821 .at(DoubleTelemetryField::ModulesDriveI)
822 .SetDefaultValue(driveGains.kI);
823 m_config.GetDoubleFields()
824 .at(DoubleTelemetryField::ModulesDriveD)
825 .SetDefaultValue(driveGains.kD);
826
827 auto* azimuthMotor = modules[0]->GetAzimuthMotorController();
828 auto azimuthGains =
829 azimuthMotor->GetConfig().GetSlotGains(azimuthMotor->GetClosedLoopControllerSlot());
830 m_config.GetDoubleFields()
831 .at(DoubleTelemetryField::ModulesAzimuthP)
832 .SetDefaultValue(azimuthGains.kP);
833 m_config.GetDoubleFields()
834 .at(DoubleTelemetryField::ModulesAzimuthI)
835 .SetDefaultValue(azimuthGains.kI);
836 m_config.GetDoubleFields()
837 .at(DoubleTelemetryField::ModulesAzimuthD)
838 .SetDefaultValue(azimuthGains.kD);
839
840 if (auto driveFF = driveMotor->GetConfig().GetSimpleFeedforward(
841 driveMotor->GetClosedLoopControllerSlot())) {
842 m_config.GetDoubleFields()
843 .at(DoubleTelemetryField::ModulesDriveKs)
844 .SetDefaultValue(driveFF->GetKs().value());
845 m_config.GetDoubleFields()
846 .at(DoubleTelemetryField::ModulesDriveKv)
847 .SetDefaultValue(driveFF->GetKv().value());
848 m_config.GetDoubleFields()
849 .at(DoubleTelemetryField::ModulesDriveKa)
850 .SetDefaultValue(driveFF->GetKa().value());
851 }
852 if (auto azimuthFF = azimuthMotor->GetConfig().GetSimpleFeedforward(
853 azimuthMotor->GetClosedLoopControllerSlot())) {
854 m_config.GetDoubleFields()
855 .at(DoubleTelemetryField::ModulesAzimuthKs)
856 .SetDefaultValue(azimuthFF->GetKs().value());
857 m_config.GetDoubleFields()
858 .at(DoubleTelemetryField::ModulesAzimuthKv)
859 .SetDefaultValue(azimuthFF->GetKv().value());
860 m_config.GetDoubleFields()
861 .at(DoubleTelemetryField::ModulesAzimuthKa)
862 .SetDefaultValue(azimuthFF->GetKa().value());
863 }
864 }
865 }
866
867 for (auto& [field, dt] : m_config.GetDoubleFields()) {
868 if (!dt.IsEnabled()) continue;
869 if (nt4Enabled) dt.SetupNetworkTables(m_dataTable, m_tuningTable);
870 if (dataLogName) dt.SetupDataLog(*dataLogName);
871 }
872 for (auto& [field, bt] : m_config.GetBoolFields()) {
873 if (!bt.IsEnabled()) continue;
874 if (nt4Enabled) bt.SetupNetworkTables(m_dataTable, m_tuningTable);
875 if (dataLogName) bt.SetupDataLog(*dataLogName);
876 }
877 for (auto& [field, stt] : m_config.GetPoseFields()) {
878 if (!stt.IsEnabled()) continue;
879 if (nt4Enabled) stt.SetupNetworkTables(m_dataTable, m_tuningTable);
880 if (dataLogName) stt.SetupDataLog(*dataLogName);

951template <std::size_t NumModules>
952void SwerveDriveTelemetry::ApplyTuningValues(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
953 bool nt4Enabled = m_config.GetNT4Enabled();
954
955 auto& boolFields = m_config.GetBoolFields();
956 auto& autoAlignBt = boolFields.at(BooleanTelemetryField::AutoAlignEnabled);
957 auto& driveTuningBt = boolFields.at(BooleanTelemetryField::ModulesDriveTuningEnabled);
958 auto& azimuthTuningBt = boolFields.at(BooleanTelemetryField::ModulesAzimuthTuningEnabled);
959 auto& driveInPlaceBt = boolFields.at(BooleanTelemetryField::ModulesDriveInPlace);
960
961 bool autoAlignOn = autoAlignBt.IsEnabled() && nt4Enabled && autoAlignBt.Get();
962 bool driveTuningOn = driveTuningBt.IsEnabled() && nt4Enabled && driveTuningBt.Get();
963 bool azimuthTuningOn = azimuthTuningBt.IsEnabled() && nt4Enabled && azimuthTuningBt.Get();
964 bool driveInPlaceOn = driveInPlaceBt.IsEnabled() && nt4Enabled && driveInPlaceBt.Get();
965
966 // Only one tuning mode may drive the chassis at a time; enforce priority order
967 // (auto-align > drive tuning > azimuth tuning) and force the losers back off in
968 // NetworkTables so the dashboard reflects what's actually happening.
969 if (autoAlignOn) {
970 if (driveTuningOn) {
971 driveTuningBt.ForceSet(false);
972 driveTuningOn = false;
973 }
974 if (azimuthTuningOn) {
975 azimuthTuningBt.ForceSet(false);
976 azimuthTuningOn = false;
977 }
978 } else if (driveTuningOn && azimuthTuningOn) {
979 azimuthTuningBt.ForceSet(false);
980 azimuthTuningOn = false;
981 }
982
983 if (autoAlignOn) {
984 auto& doubleFields = m_config.GetDoubleFields();
985 double x = doubleFields.at(DoubleTelemetryField::AutoAlignPoseX).Get();
986 double y = doubleFields.at(DoubleTelemetryField::AutoAlignPoseY).Get();
987 double rot = doubleFields.at(DoubleTelemetryField::AutoAlignPoseRotation).Get();
988 frc::Pose2d targetPose{units::meter_t{x}, units::meter_t{y},
989 frc::Rotation2d{units::degree_t{rot}}};
990 drive->SetRobotRelativeChassisSpeeds(drive->DriveToPoseSetpoint(targetPose));
991 }
992
993 if (driveTuningBt.IsEnabled() && nt4Enabled) {
994 auto& modules = drive->GetConfig().GetModules();
995 if (driveTuningOn) {
996 units::meters_per_second_t velocity{
997 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesDriveVelocity).Get()};
998 wpi::array<frc::SwerveModuleState, NumModules> states{wpi::empty_array};
999 for (size_t i = 0; i < NumModules; ++i) {
1000 if (driveInPlaceOn) {
1001 // Point each module tangent to its position around the robot center so a positive
1002 // velocity spins the robot counter-clockwise (WPILib's positive rotation direction).
1003 auto moduleLocation = *modules[i]->GetConfig().GetLocation();
1004 auto tangentAngle = moduleLocation.Angle() + frc::Rotation2d{units::degree_t{90}};
1005 states[i] = frc::SwerveModuleState{velocity, tangentAngle};
1006 } else {
1007 states[i] = frc::SwerveModuleState{velocity, frc::Rotation2d{}};
1008 }
1009 }
1010 drive->SetSwerveModuleStates(states);
1011 } else if (!autoAlignOn) {
1012 // Drive tuning is off (and auto-align isn't driving the chassis instead); make sure the
1013 // modules don't keep spinning at whatever velocity was last commanded while it was on.
1014 wpi::array<frc::SwerveModuleState, NumModules> states{wpi::empty_array};
1015 for (size_t i = 0; i < NumModules; ++i) {
1016 states[i] =
1017 frc::SwerveModuleState{units::meters_per_second_t{0}, modules[i]->GetState().angle};
1018 }
1019 drive->SetSwerveModuleStates(states);
1020 }
1021 }
1022
1023 if (azimuthTuningOn) {
1024 frc::Rotation2d angle{units::degree_t{
1025 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesAzimuthAngle).Get()}};
1026 wpi::array<frc::SwerveModuleState, NumModules> states{wpi::empty_array};
1027 for (size_t i = 0; i < NumModules; ++i) {
1028 states[i] = frc::SwerveModuleState{units::meters_per_second_t{0}, angle};
1029 }
1030 drive->SetSwerveModuleStates(states);
1031 }
1032
1033 // Copies of the configured controllers; gain fields for missing controllers are disabled.
1034 std::optional<frc::PIDController> translationPID;
1035 if (auto pid = drive->GetConfig().GetTranslationPID()) translationPID = pid->get();
1036 std::optional<frc::PIDController> rotationPID;
1037 if (auto pid = drive->GetConfig().GetRotationPID()) rotationPID = pid->get();
1038 for (auto& [field, dt] : m_config.GetDoubleFields()) {
1039 if (!dt.IsTunable()) continue;
1040 switch (field) {
1041 case DoubleTelemetryField::TranslationP:
1042 if (translationPID) {
1043 translationPID->SetP(dt.Get());
1044 drive->SetTranslationPID(*translationPID);
1045 }
1046 break;
1047 case DoubleTelemetryField::TranslationI:
1048 if (translationPID) {
1049 translationPID->SetI(dt.Get());
1050 drive->SetTranslationPID(*translationPID);
1051 }
1052 break;
1053 case DoubleTelemetryField::TranslationD:
1054 if (translationPID) {
1055 translationPID->SetD(dt.Get());
1056 drive->SetTranslationPID(*translationPID);
1057 }
1058 break;
1059 case DoubleTelemetryField::RotationP:
1060 if (rotationPID) {
1061 rotationPID->SetP(dt.Get());
1062 drive->SetRotationPID(*rotationPID);
1063 }
1064 break;
1065 case DoubleTelemetryField::RotationI:
1066 if (rotationPID) {
1067 rotationPID->SetI(dt.Get());
1068 drive->SetRotationPID(*rotationPID);
1069 }
1070 break;
1071 case DoubleTelemetryField::RotationD:
1072 if (rotationPID) {
1073 rotationPID->SetD(dt.Get());
1074 drive->SetRotationPID(*rotationPID);
1075 }
1076 break;
1077 case DoubleTelemetryField::ModulesDriveP:
1078 for (auto* module : drive->GetConfig().GetModules()) {
1079 module->GetDriveMotorController()->SetKp(dt.Get());
1080 }
1081 break;
1082 case DoubleTelemetryField::ModulesDriveI:
1083 for (auto* module : drive->GetConfig().GetModules()) {
1084 module->GetDriveMotorController()->SetKi(dt.Get());
1085 }
1086 break;
1087 case DoubleTelemetryField::ModulesDriveD:
1088 for (auto* module : drive->GetConfig().GetModules()) {
1089 module->GetDriveMotorController()->SetKd(dt.Get());
1090 }
1091 break;
1092 case DoubleTelemetryField::ModulesAzimuthP:
1093 for (auto* module : drive->GetConfig().GetModules()) {
1094 module->GetAzimuthMotorController()->SetKp(dt.Get());
1095 }
1096 break;
1097 case DoubleTelemetryField::ModulesAzimuthI:
1098 for (auto* module : drive->GetConfig().GetModules()) {
1099 module->GetAzimuthMotorController()->SetKi(dt.Get());
1100 }
1101 break;
1102 case DoubleTelemetryField::ModulesAzimuthD:
1103 for (auto* module : drive->GetConfig().GetModules()) {
1104 module->GetAzimuthMotorController()->SetKd(dt.Get());
1105 }
1106 break;
1107 case DoubleTelemetryField::ModulesDriveKs:
1108 for (auto* module : drive->GetConfig().GetModules()) {
1109 module->GetDriveMotorController()->SetKs(dt.Get());
1110 }
1111 break;
1112 case DoubleTelemetryField::ModulesDriveKv:
1113 for (auto* module : drive->GetConfig().GetModules()) {
1114 module->GetDriveMotorController()->SetKv(dt.Get());
1115 }
1116 break;
1117 case DoubleTelemetryField::ModulesDriveKa:
1118 for (auto* module : drive->GetConfig().GetModules()) {
1119 module->GetDriveMotorController()->SetKa(dt.Get());
1120 }
1121 break;
1122 case DoubleTelemetryField::ModulesAzimuthKs:
1123 for (auto* module : drive->GetConfig().GetModules()) {
1124 module->GetAzimuthMotorController()->SetKs(dt.Get());
1125 }
1126 break;
1127 case DoubleTelemetryField::ModulesAzimuthKv:
1128 for (auto* module : drive->GetConfig().GetModules()) {
1129 module->GetAzimuthMotorController()->SetKv(dt.Get());
1130 }
1131 break;
1132 case DoubleTelemetryField::ModulesAzimuthKa:
1133 for (auto* module : drive->GetConfig().GetModules()) {
1134 module->GetAzimuthMotorController()->SetKa(dt.Get());
1135 }
1136 break;
1137 default:
1138 break;
1139 }
1140 }
