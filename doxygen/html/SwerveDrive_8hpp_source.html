192 frc2::CommandPtr DriveToPose(frc::Pose2d pose) {
193 return frc2::cmd::RunOnce([this] {
194 ResetTranslationPID();
195 ResetAzimuthPID();
196 })
197 .AndThen(Drive([this, pose]() -> frc::ChassisSpeeds {
198 auto& azimuthPID = m_config->GetRotationPID();
199 auto& translationPID = m_config->GetTranslationPID();
200 auto currentPose = GetPose();
201 auto distance = GetDistanceFromPose(pose);
202 auto translationScalar = translationPID.Calculate(distance.value(), 0.0);
203 auto poseDiff = currentPose.RelativeTo(pose);
204 return frc::ChassisSpeeds::FromFieldRelativeSpeeds(
205 frc::ChassisSpeeds{
206 units::meters_per_second_t{poseDiff.X().value() * translationScalar},
207 units::meters_per_second_t{poseDiff.Y().value() * translationScalar},
208 units::radians_per_second_t{
209 azimuthPID.Calculate(currentPose.Rotation().Radians().value(),
210 pose.Rotation().Radians().value())}},
211 frc::Rotation2d{units::radian_t{GetGyroAngle()}});
212 }))
213 .WithName("Drive to Pose");
214 }

347 void UpdateTelemetry() {
348 UpdatePoseEstimator();
349 auto pose = GetPose();
350 auto currentStates = GetModuleStates();
351
352 m_gyroPublisher(units::degree_t{GetGyroAngle()}.value());
353 m_desiredModuleStatesPublisher(m_desiredModuleStates);
354 m_currentModuleStatesPublisher(currentStates);
355 m_posePublisher(pose);
356 m_desiredRobotRelChassisSpeedsPublisher(m_desiredChassisSpeeds);
357 m_currentRobotRelChassisSpeedsPublisher(GetRobotRelativeSpeed());
358 m_fieldRelChassisSpeedsPublisher(GetFieldRelativeSpeed());
359
360 for (auto* mod : m_config->GetModules()) {
361 mod->UpdateTelemetry();
362 }
363
364 m_field2d.SetRobotPose(pose);
365 std::vector<frc::Pose2d> modulePoses;
366 modulePoses.reserve(NumModules);
367 for (size_t i = 0; i < NumModules; ++i) {
368 auto* mod = m_config->GetModules()[i];
369 auto location = *mod->GetConfig().GetLocation();
370 auto moduleTranslation = pose.Translation() + location.RotateBy(pose.Rotation());
371 auto moduleHeading = pose.Rotation() + currentStates[i].angle;
372 modulePoses.push_back(frc::Pose2d{moduleTranslation, moduleHeading});
373 }
374 m_field2d.GetObject("modules")->SetPoses(modulePoses);
375 }

384 void SimIterate() {
385 if (!m_simTimer.IsRunning()) m_simTimer.Start();
386 for (auto* mod : m_config->GetModules()) {
387 mod->SimIterate();
388 }
389 auto desired = m_kinematics.ToChassisSpeeds(m_desiredModuleStates);
390
391 auto dt = m_simTimer.Get();
392 frc::Twist2d twist{desired.vx * dt, desired.vy * dt, desired.omega * dt};
393 m_simPose = m_simPose.Exp(twist);
394
395 auto speeds = m_kinematics.ToChassisSpeeds(GetModuleStates());
396 m_simGyroAngle +=
397 units::degree_t{units::degrees_per_second_t{speeds.omega}.value() * dt.value()};
398 m_simTimer.Reset();
399 }

578 static frc::SwerveDriveKinematics<NumModules> BuildKinematics(const SwerveDriveConfig& config) {
579 assert(config.GetModules().size() == NumModules);
580 wpi::array<frc::Translation2d, NumModules> locations{wpi::empty_array};
