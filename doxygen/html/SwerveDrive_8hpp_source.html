278 wpi::math::ChassisVelocities DriveToPoseSetpoint(wpi::math::Pose2d targetPose,
279 wpi::math::Pose2d currentPose,
280 wpi::math::Rotation2d heading) {
281 auto rotationPIDRef = m_config->GetRotationPID();
282 if (!rotationPIDRef) {
283 throw exceptions::SwerveDriveConfigurationException(
284 "No rotation PID controller configured, cannot drive to pose. Use "
285 "SwerveDriveConfig::WithRotationController(PIDController) to fix this error.");
286 }
287 auto translationPIDRef = m_config->GetTranslationPID();
288 if (!translationPIDRef) {
289 throw exceptions::SwerveDriveConfigurationException(
290 "No translation PID controller configured, cannot drive to pose. Use "
291 "SwerveDriveConfig::WithTranslationController(PIDController) to fix this error.");
292 }
293 auto& rotationPID = rotationPIDRef->get();
294 auto& translationPID = translationPIDRef->get();
295 // Plain field-frame translation delta (not RelativeTo/operator-, which expresses the result
296 // in targetPose's rotated frame and would skew the commanded direction whenever targetPose's
297 // heading is non-zero).
298 auto toTarget = targetPose.Translation() - currentPose.Translation();
299 double distance = toTarget.Norm().value();
300 // The controller sees the distance as its measurement with a setpoint of 0, so its output is
301 // negative while there is distance left: the speed toward the target is the negative of it.
302 double speed = -translationPID.Calculate(distance, 0.0);
303 // Drive at that speed along the unit vector toward the target.
304 double vx = distance > 1e-9 ? toTarget.X().value() / distance * speed : 0.0;
305 double vy = distance > 1e-9 ? toTarget.Y().value() / distance * speed : 0.0;
306 return (wpi::math::ChassisVelocities{
307 wpi::units::meters_per_second_t{vx}, wpi::units::meters_per_second_t{vy},
308 wpi::units::radians_per_second_t{
309 rotationPID.Calculate(currentPose.Rotation().Radians().value(),
310 targetPose.Rotation().Radians().value())}})
311 .ToRobotRelative(heading);
312 }

347 wpi::util::array<wpi::math::SwerveModuleVelocity, NumModules>
348 GetStateFromRobotRelativeChassisSpeeds(wpi::math::ChassisVelocities robotRelativeSpeeds) {
349 robotRelativeSpeeds = m_config->OptimizeRobotRelativeChassisSpeeds(robotRelativeSpeeds);
350 auto cor = m_config->GetCenterOfRotation();
351 auto states = cor ? m_kinematics.ToSwerveModuleVelocities(robotRelativeSpeeds, *cor)
352 : m_kinematics.ToSwerveModuleVelocities(robotRelativeSpeeds);
353 // Scale the wheels down together when one is asked to go faster than it can.
354 if (auto maxModuleSpeed = m_config->GetMaximumModuleLinearVelocity()) {
355 auto maxLinear = m_config->GetMaximumChassisLinearVelocity();
356 auto maxAngular = m_config->GetMaximumChassisAngularVelocity();
357 if (maxLinear && maxAngular) {
358 states = wpi::math::SwerveDriveKinematics<NumModules>::DesaturateWheelVelocities(
359 states, robotRelativeSpeeds, *maxModuleSpeed, *maxLinear,
360 wpi::units::radians_per_second_t{*maxAngular});
361 } else {
362 states = wpi::math::SwerveDriveKinematics<NumModules>::DesaturateWheelVelocities(
363 states, *maxModuleSpeed);
364 }
365 }
366 return states;
367 }

398 void SetRobotRelativeChassisSpeeds(wpi::math::ChassisVelocities robotRelativeSpeeds,
399 std::span<const wpi::units::newton_t> feedforwardForces) {
400 m_desiredChassisSpeeds = robotRelativeSpeeds;

407 throw std::invalid_argument("feedforwardForces must be empty or have one entry per module (" +
408 std::to_string(NumModules) + "), in FL, FR, BL, BR order.");
409 }
410 SetSwerveModuleStates(states, feedforwardForces);

513 void UpdateTelemetry() {
514 UpdatePoseEstimator();
515 auto pose = GetPose();
516 auto currentStates = GetModuleStates();
517
518 m_swerveTelemetry->Publish(this);
519
520 for (auto* mod : m_config->GetModules()) {
521 mod->UpdateTelemetry();
522 }
523 m_telemetry.UpdateLoopTime();
524
525 m_field2d.SetRobotPose(pose);
526 std::vector<wpi::math::Pose2d> modulePoses;
527 modulePoses.reserve(NumModules);
528 for (size_t i = 0; i < NumModules; ++i) {
529 auto* mod = m_config->GetModules()[i];
530 auto location = *mod->GetConfig().GetLocation();
531 auto moduleTranslation = pose.Translation() + location.RotateBy(pose.Rotation());
532 auto moduleHeading = pose.Rotation() + currentStates[i].angle;
533 modulePoses.push_back(wpi::math::Pose2d{moduleTranslation, moduleHeading});
534 }
535 m_field2d.GetObject("modules")->SetPoses(modulePoses);
536 }
537
545 void SimIterate() {
546 if (!m_simTimer.IsRunning()) m_simTimer.Start();
547 for (auto* mod : m_config->GetModules()) {
548 mod->SimIterate();
549 }
550 auto desired = m_kinematics.ToChassisVelocities(m_desiredModuleStates);
551
552 auto dt = m_simTimer.Get();
553 wpi::math::Twist2d twist{desired.vx * dt, desired.vy * dt, desired.omega * dt};
554 m_simPose = m_simPose + twist.Exp();
555
556 auto speeds = m_kinematics.ToChassisVelocities(GetModuleStates());
557 m_simGyroAngle +=
558 wpi::units::degree_t{wpi::units::degrees_per_second_t{speeds.omega}.value() * dt.value()};
559 m_simTimer.Reset();
560 }

856// SwerveDriveTelemetry::SetupTelemetry()/Publish()/ApplyTuningValues() are templated on the
857// SwerveDrive's module count so that SwerveDriveTelemetry itself need not be a template. Their
858// bodies live here after SwerveDrive<NumModules> is fully defined above because this is the
859// only place both SwerveDriveTelemetry's private members and SwerveDrive<NumModules>'s interface
860// are simultaneously visible.
861
862template <std::size_t NumModules>
863void SwerveDriveTelemetry::SetupTelemetry(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
864 auto driveName = drive->GetName();
865 for (auto* module : drive->GetConfig().GetModules()) {
866 module->SetupTelemetry(driveName);
867 }
868
869 auto inst = wpi::nt::NetworkTableInstance::GetDefault();
870 m_dataTable = inst.GetTable("Mechanisms")->GetSubTable(driveName);
871 m_tuningTable = inst.GetTable("Tuning")->GetSubTable(driveName);
872
873 bool nt4Enabled = m_config.GetNT4Enabled();
874 auto dataLogName = m_config.GetDataLogName();
875
876 {
877 // Only publish the drive to pose gains and auto-align fields for controllers that exist.
878 auto translationPID = drive->GetConfig().GetTranslationPID();
879 auto rotationPID = drive->GetConfig().GetRotationPID();
880 auto& doubleFields = m_config.GetDoubleFields();
881 if (translationPID) {
882 doubleFields.at(DoubleTelemetryField::TranslationP)
883 .SetDefaultValue(translationPID->get().GetP());
884 doubleFields.at(DoubleTelemetryField::TranslationI)
885 .SetDefaultValue(translationPID->get().GetI());
886 doubleFields.at(DoubleTelemetryField::TranslationD)
887 .SetDefaultValue(translationPID->get().GetD());
888 } else {
889 doubleFields.at(DoubleTelemetryField::TranslationP).Disable();
890 doubleFields.at(DoubleTelemetryField::TranslationI).Disable();
891 doubleFields.at(DoubleTelemetryField::TranslationD).Disable();
892 }
893 if (rotationPID) {
894 doubleFields.at(DoubleTelemetryField::RotationP).SetDefaultValue(rotationPID->get().GetP());
895 doubleFields.at(DoubleTelemetryField::RotationI).SetDefaultValue(rotationPID->get().GetI());
896 doubleFields.at(DoubleTelemetryField::RotationD).SetDefaultValue(rotationPID->get().GetD());
897 } else {
898 doubleFields.at(DoubleTelemetryField::RotationP).Disable();
899 doubleFields.at(DoubleTelemetryField::RotationI).Disable();
900 doubleFields.at(DoubleTelemetryField::RotationD).Disable();

903 if (!translationPID || !rotationPID) {
904 doubleFields.at(DoubleTelemetryField::AutoAlignPoseX).Disable();
905 doubleFields.at(DoubleTelemetryField::AutoAlignPoseY).Disable();
906 doubleFields.at(DoubleTelemetryField::AutoAlignPoseRotation).Disable();
907 m_config.GetBoolFields().at(BooleanTelemetryField::AutoAlignEnabled).Disable();
908 }

915 m_config.GetDoubleFields()

919 .at(DoubleTelemetryField::ModulesDriveI)

928 .at(DoubleTelemetryField::ModulesDriveKv)

930 m_config.GetDoubleFields()
931 .at(DoubleTelemetryField::ModulesDriveKa)
932 .SetDefaultValue(driveGains.kA);

935 auto azimuthGains =
936 azimuthMotor->GetConfig().GetSlotGains(azimuthMotor->GetClosedLoopControllerSlot());
937 m_config.GetDoubleFields()

940 m_config.GetDoubleFields()
941 .at(DoubleTelemetryField::ModulesAzimuthI)

951 .SetDefaultValue(azimuthGains.kV);

953 .at(DoubleTelemetryField::ModulesAzimuthKa)
954 .SetDefaultValue(azimuthGains.kA);

957
958 for (auto& [field, dt] : m_config.GetDoubleFields()) {
959 if (!dt.IsEnabled()) continue;
960 if (nt4Enabled) dt.SetupNetworkTables(m_dataTable, m_tuningTable);
961 if (dataLogName) dt.SetupDataLog(*dataLogName);

963 for (auto& [field, bt] : m_config.GetBoolFields()) {

1042template <std::size_t NumModules>
1043void SwerveDriveTelemetry::ApplyTuningValues(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
1044 bool nt4Enabled = m_config.GetNT4Enabled();
1045
1046 auto& boolFields = m_config.GetBoolFields();
1047 auto& autoAlignBt = boolFields.at(BooleanTelemetryField::AutoAlignEnabled);
1048 auto& driveTuningBt = boolFields.at(BooleanTelemetryField::ModulesDriveTuningEnabled);
1049 auto& azimuthTuningBt = boolFields.at(BooleanTelemetryField::ModulesAzimuthTuningEnabled);
1050 auto& driveInPlaceBt = boolFields.at(BooleanTelemetryField::ModulesDriveInPlace);
1051
1052 bool autoAlignOn = autoAlignBt.IsEnabled() && nt4Enabled && autoAlignBt.Get();
1053 bool driveTuningOn = driveTuningBt.IsEnabled() && nt4Enabled && driveTuningBt.Get();
1054 bool azimuthTuningOn = azimuthTuningBt.IsEnabled() && nt4Enabled && azimuthTuningBt.Get();
1055 bool driveInPlaceOn = driveInPlaceBt.IsEnabled() && nt4Enabled && driveInPlaceBt.Get();
1056
1057 // Only one tuning mode may drive the chassis at a time; enforce priority order
1058 // (auto-align > drive tuning > azimuth tuning) and force the losers back off in
1059 // NetworkTables so the dashboard reflects what's actually happening.
1060 if (autoAlignOn) {
1061 if (driveTuningOn) {
1062 driveTuningBt.ForceSet(false);
1063 driveTuningOn = false;
1064 }
1065 if (azimuthTuningOn) {
1066 azimuthTuningBt.ForceSet(false);
1067 azimuthTuningOn = false;
1068 }
1069 } else if (driveTuningOn && azimuthTuningOn) {
1070 azimuthTuningBt.ForceSet(false);
1071 azimuthTuningOn = false;
1072 }
1073
1074 if (autoAlignOn) {
1075 auto& doubleFields = m_config.GetDoubleFields();
1076 double x = doubleFields.at(DoubleTelemetryField::AutoAlignPoseX).Get();
1077 double y = doubleFields.at(DoubleTelemetryField::AutoAlignPoseY).Get();
1078 double rot = doubleFields.at(DoubleTelemetryField::AutoAlignPoseRotation).Get();
1079 wpi::math::Pose2d targetPose{wpi::units::meter_t{x}, wpi::units::meter_t{y},
1080 wpi::math::Rotation2d{wpi::units::degree_t{rot}}};
1081 drive->SetRobotRelativeChassisSpeeds(drive->DriveToPoseSetpoint(targetPose));
1082 }
1083
1084 if (driveTuningBt.IsEnabled() && nt4Enabled) {
1085 auto& modules = drive->GetConfig().GetModules();
1086 if (driveTuningOn) {
1087 wpi::units::meters_per_second_t velocity{
1088 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesDriveVelocity).Get()};
1089 wpi::util::array<wpi::math::SwerveModuleVelocity, NumModules> states{wpi::util::empty_array};
1090 for (size_t i = 0; i < NumModules; ++i) {
1091 if (driveInPlaceOn) {
1092 // Point each module tangent to its position around the robot center so a positive
1093 // velocity spins the robot counter-clockwise (WPILib's positive rotation direction).
1094 auto moduleLocation = *modules[i]->GetConfig().GetLocation();
1095 auto tangentAngle = moduleLocation.Angle().value_or(wpi::math::Rotation2d{}) +
1096 wpi::math::Rotation2d{wpi::units::degree_t{90}};
1097 states[i] = wpi::math::SwerveModuleVelocity{velocity, tangentAngle};
1098 } else {
1099 states[i] = wpi::math::SwerveModuleVelocity{velocity, wpi::math::Rotation2d{}};
1100 }
1101 }
1102 drive->SetSwerveModuleStates(states);
1103 } else if (!autoAlignOn) {
1104 // Drive tuning is off (and auto-align isn't driving the chassis instead); make sure the
1105 // modules don't keep spinning at whatever velocity was last commanded while it was on.
1106 wpi::util::array<wpi::math::SwerveModuleVelocity, NumModules> states{wpi::util::empty_array};
1107 for (size_t i = 0; i < NumModules; ++i) {
1108 states[i] = wpi::math::SwerveModuleVelocity{wpi::units::meters_per_second_t{0},
1109 modules[i]->GetState().angle};
1110 }
1111 drive->SetSwerveModuleStates(states);
1112 }
1113 }
1114
1115 if (azimuthTuningOn) {
1116 // Zero speed states would be held where the wheels are by the velocity deadband, so turn the
1117 // wheels without driving them.
1118 drive->SetAzimuthAngles(wpi::math::Rotation2d{wpi::units::degree_t{
1119 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesAzimuthAngle).Get()}});
1120 }
1121
1122 // Copies of the configured controllers; gain fields for missing controllers are disabled.
1123 std::optional<wpi::math::PIDController> translationPID;
1124 if (auto pid = drive->GetConfig().GetTranslationPID()) translationPID = pid->get();
1125 std::optional<wpi::math::PIDController> rotationPID;
1126 if (auto pid = drive->GetConfig().GetRotationPID()) rotationPID = pid->get();
1127 for (auto& [field, dt] : m_config.GetDoubleFields()) {
1128 if (!dt.IsTunable()) continue;
1129 switch (field) {
1130 case DoubleTelemetryField::TranslationP:
1131 if (translationPID) {
1132 translationPID->SetP(dt.Get());
1133 drive->SetTranslationPID(*translationPID);
1134 }
1135 break;
1136 case DoubleTelemetryField::TranslationI:
1137 if (translationPID) {
1138 translationPID->SetI(dt.Get());
1139 drive->SetTranslationPID(*translationPID);
1140 }
1141 break;
1142 case DoubleTelemetryField::TranslationD:
1143 if (translationPID) {
1144 translationPID->SetD(dt.Get());
1145 drive->SetTranslationPID(*translationPID);
1146 }
1147 break;
1148 case DoubleTelemetryField::RotationP:
1149 if (rotationPID) {
1150 rotationPID->SetP(dt.Get());
1151 drive->SetRotationPID(*rotationPID);
1152 }
1153 break;
1154 case DoubleTelemetryField::RotationI:
1155 if (rotationPID) {
1156 rotationPID->SetI(dt.Get());
1157 drive->SetRotationPID(*rotationPID);
1158 }
1159 break;
1160 case DoubleTelemetryField::RotationD:
1161 if (rotationPID) {
1162 rotationPID->SetD(dt.Get());
1163 drive->SetRotationPID(*rotationPID);
1164 }
1165 break;
1166 case DoubleTelemetryField::ModulesDriveP:
1167 for (auto* module : drive->GetConfig().GetModules()) {
1168 module->GetDriveMotorController()->SetKp(dt.Get());
1169 }
1170 break;
1171 case DoubleTelemetryField::ModulesDriveI:
1172 for (auto* module : drive->GetConfig().GetModules()) {
1173 module->GetDriveMotorController()->SetKi(dt.Get());
1174 }
1175 break;
1176 case DoubleTelemetryField::ModulesDriveD:
1177 for (auto* module : drive->GetConfig().GetModules()) {
1178 module->GetDriveMotorController()->SetKd(dt.Get());
1179 }
1180 break;
1181 case DoubleTelemetryField::ModulesAzimuthP:
1182 for (auto* module : drive->GetConfig().GetModules()) {
1183 module->GetAzimuthMotorController()->SetKp(dt.Get());
1184 }
1185 break;
1186 case DoubleTelemetryField::ModulesAzimuthI:
1187 for (auto* module : drive->GetConfig().GetModules()) {
1188 module->GetAzimuthMotorController()->SetKi(dt.Get());
1189 }
1190 break;
1191 case DoubleTelemetryField::ModulesAzimuthD:
1192 for (auto* module : drive->GetConfig().GetModules()) {
1193 module->GetAzimuthMotorController()->SetKd(dt.Get());
1194 }
1195 break;
1196 case DoubleTelemetryField::ModulesDriveKs:
1197 for (auto* module : drive->GetConfig().GetModules()) {
1198 module->GetDriveMotorController()->SetKs(dt.Get());
1199 }
1200 break;
1201 case DoubleTelemetryField::ModulesDriveKv:
1202 for (auto* module : drive->GetConfig().GetModules()) {
1203 module->GetDriveMotorController()->SetKv(dt.Get());
1204 }
1205 break;
1206 case DoubleTelemetryField::ModulesDriveKa:
1207 for (auto* module : drive->GetConfig().GetModules()) {
1208 module->GetDriveMotorController()->SetKa(dt.Get());
1209 }
1210 break;
1211 case DoubleTelemetryField::ModulesAzimuthKs:
1212 for (auto* module : drive->GetConfig().GetModules()) {
1213 module->GetAzimuthMotorController()->SetKs(dt.Get());
1214 }
1215 break;
1216 case DoubleTelemetryField::ModulesAzimuthKv:
1217 for (auto* module : drive->GetConfig().GetModules()) {
1218 module->GetAzimuthMotorController()->SetKv(dt.Get());
1219 }
1220 break;
1221 case DoubleTelemetryField::ModulesAzimuthKa:
1222 for (auto* module : drive->GetConfig().GetModules()) {
1223 module->GetAzimuthMotorController()->SetKa(dt.Get());
1224 }
1225 break;
1226 default:
1227 break;
1228 }
1229 }
1230}
