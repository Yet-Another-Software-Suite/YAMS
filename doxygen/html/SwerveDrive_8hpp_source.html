198 frc::ChassisSpeeds DriveToPoseSetpoint(frc::Pose2d targetPose) {
199 auto& rotationPID = m_config->GetRotationPID();
200 auto& translationPID = m_config->GetTranslationPID();
201 auto distance = GetDistanceFromPose(targetPose);
202 auto translationScalar = translationPID.Calculate(distance.value(), 0.0);
203 auto currentPose = GetPose();
204 // Plain field-frame translation delta (not RelativeTo/operator-, which expresses the result
205 // in targetPose's rotated frame and would skew the commanded direction whenever targetPose's
206 // heading is non-zero).
207 auto translationDiff = currentPose.Translation() - targetPose.Translation();
208 return frc::ChassisSpeeds::FromFieldRelativeSpeeds(
209 frc::ChassisSpeeds{
210 units::meters_per_second_t{translationDiff.X().value() * translationScalar},
211 units::meters_per_second_t{translationDiff.Y().value() * translationScalar},
212 units::radians_per_second_t{
213 rotationPID.Calculate(currentPose.Rotation().Radians().value(),
214 targetPose.Rotation().Radians().value())}},
215 frc::Rotation2d{units::radian_t{GetGyroAngle()}});
216 }

349 void UpdateTelemetry() {
350 UpdatePoseEstimator();
351 auto pose = GetPose();
352 auto currentStates = GetModuleStates();
353
354 m_swerveTelemetry->Publish(this);
355
356 for (auto* mod : m_config->GetModules()) {
357 mod->UpdateTelemetry();
358 }
359 m_telemetry.UpdateLoopTime();
360
361 m_field2d.SetRobotPose(pose);
362 std::vector<frc::Pose2d> modulePoses;
363 modulePoses.reserve(NumModules);
364 for (size_t i = 0; i < NumModules; ++i) {
365 auto* mod = m_config->GetModules()[i];
366 auto location = *mod->GetConfig().GetLocation();
367 auto moduleTranslation = pose.Translation() + location.RotateBy(pose.Rotation());
368 auto moduleHeading = pose.Rotation() + currentStates[i].angle;
369 modulePoses.push_back(frc::Pose2d{moduleTranslation, moduleHeading});
370 }
371 m_field2d.GetObject("modules")->SetPoses(modulePoses);
372 }

381 void SimIterate() {
382 if (!m_simTimer.IsRunning()) m_simTimer.Start();
383 for (auto* mod : m_config->GetModules()) {
384 mod->SimIterate();
385 }
386 auto desired = m_kinematics.ToChassisSpeeds(m_desiredModuleStates);
387
388 auto dt = m_simTimer.Get();
389 frc::Twist2d twist{desired.vx * dt, desired.vy * dt, desired.omega * dt};
390 m_simPose = m_simPose.Exp(twist);
391
392 auto speeds = m_kinematics.ToChassisSpeeds(GetModuleStates());
393 m_simGyroAngle +=
394 units::degree_t{units::degrees_per_second_t{speeds.omega}.value() * dt.value()};
395 m_simTimer.Reset();
396 }

652template <std::size_t NumModules>
653void SwerveDriveTelemetry::SetupTelemetry(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
654 auto driveName = drive->GetName();
655 for (auto* module : drive->GetConfig().GetModules()) {
656 module->SetupTelemetry(driveName);
657 }
658
659 auto inst = nt::NetworkTableInstance::GetDefault();
660 m_dataTable = inst.GetTable("Mechanisms")->GetSubTable(driveName);
661 m_tuningTable = inst.GetTable("Tuning")->GetSubTable(driveName);
662
663 bool nt4Enabled = m_config.GetNT4Enabled();
664 auto dataLogName = m_config.GetDataLogName();
665
666 {
667 auto& translationPID = drive->GetConfig().GetTranslationPID();
668 auto& rotationPID = drive->GetConfig().GetRotationPID();
669 m_config.GetDoubleFields()
670 .at(DoubleTelemetryField::TranslationP)
671 .SetDefaultValue(translationPID.GetP());
672 m_config.GetDoubleFields()
673 .at(DoubleTelemetryField::TranslationI)
674 .SetDefaultValue(translationPID.GetI());
675 m_config.GetDoubleFields()
676 .at(DoubleTelemetryField::TranslationD)
677 .SetDefaultValue(translationPID.GetD());
678 m_config.GetDoubleFields().at(DoubleTelemetryField::RotationP).SetDefaultValue(rotationPID.GetP());
679 m_config.GetDoubleFields().at(DoubleTelemetryField::RotationI).SetDefaultValue(rotationPID.GetI());
680 m_config.GetDoubleFields().at(DoubleTelemetryField::RotationD).SetDefaultValue(rotationPID.GetD());
681
682 auto& modules = drive->GetConfig().GetModules();
683 if (!modules.empty()) {
684 auto* driveMotor = modules[0]->GetDriveMotorController();
685 auto driveGains = driveMotor->GetConfig().GetSlotGains(driveMotor->GetClosedLoopControllerSlot());
686 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesDriveP).SetDefaultValue(driveGains.kP);
687 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesDriveI).SetDefaultValue(driveGains.kI);
688 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesDriveD).SetDefaultValue(driveGains.kD);
689
690 auto* azimuthMotor = modules[0]->GetAzimuthMotorController();
691 auto azimuthGains =
692 azimuthMotor->GetConfig().GetSlotGains(azimuthMotor->GetClosedLoopControllerSlot());
693 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesAzimuthP).SetDefaultValue(azimuthGains.kP);
694 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesAzimuthI).SetDefaultValue(azimuthGains.kI);
695 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesAzimuthD).SetDefaultValue(azimuthGains.kD);
696
697 if (auto driveFF = driveMotor->GetConfig().GetSimpleFeedforward(
698 driveMotor->GetClosedLoopControllerSlot())) {
699 m_config.GetDoubleFields()
700 .at(DoubleTelemetryField::ModulesDriveKs)
701 .SetDefaultValue(driveFF->GetKs().value());
702 m_config.GetDoubleFields()
703 .at(DoubleTelemetryField::ModulesDriveKv)
704 .SetDefaultValue(driveFF->GetKv().value());
705 m_config.GetDoubleFields()
706 .at(DoubleTelemetryField::ModulesDriveKa)
707 .SetDefaultValue(driveFF->GetKa().value());
708 }
709 if (auto azimuthFF = azimuthMotor->GetConfig().GetSimpleFeedforward(
710 azimuthMotor->GetClosedLoopControllerSlot())) {
711 m_config.GetDoubleFields()
712 .at(DoubleTelemetryField::ModulesAzimuthKs)
713 .SetDefaultValue(azimuthFF->GetKs().value());
714 m_config.GetDoubleFields()
715 .at(DoubleTelemetryField::ModulesAzimuthKv)
716 .SetDefaultValue(azimuthFF->GetKv().value());
717 m_config.GetDoubleFields()
718 .at(DoubleTelemetryField::ModulesAzimuthKa)
719 .SetDefaultValue(azimuthFF->GetKa().value());
720 }
721 }
722 }
723
724 for (auto& [field, dt] : m_config.GetDoubleFields()) {
725 if (!dt.IsEnabled()) continue;
726 if (nt4Enabled) dt.SetupNetworkTables(m_dataTable, m_tuningTable);
727 if (dataLogName) dt.SetupDataLog(*dataLogName);
728 }
729 for (auto& [field, bt] : m_config.GetBoolFields()) {
730 if (!bt.IsEnabled()) continue;
731 if (nt4Enabled) bt.SetupNetworkTables(m_dataTable, m_tuningTable);
732 if (dataLogName) bt.SetupDataLog(*dataLogName);
733 }
734 for (auto& [field, stt] : m_config.GetPoseFields()) {
735 if (!stt.IsEnabled()) continue;
736 if (nt4Enabled) stt.SetupNetworkTables(m_dataTable, m_tuningTable);
737 if (dataLogName) stt.SetupDataLog(*dataLogName);
738 }
739 for (auto& [field, stt] : m_config.GetChassisSpeedsFields()) {
740 if (!stt.IsEnabled()) continue;
741 if (nt4Enabled) stt.SetupNetworkTables(m_dataTable, m_tuningTable);
742 if (dataLogName) stt.SetupDataLog(*dataLogName);
743 }
744 for (auto& [field, stat] : m_config.GetModuleStatesFields()) {
745 if (!stat.IsEnabled()) continue;
746 if (nt4Enabled) stat.SetupNetworkTables(m_dataTable, m_tuningTable);
747 if (dataLogName) stat.SetupDataLog(*dataLogName);
748 }
749}

751template <std::size_t NumModules>
752void SwerveDriveTelemetry::Publish(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
753 for (auto& [field, stt] : m_config.GetPoseFields()) {
754 if (!stt.IsEnabled()) continue;
755 switch (field) {
756 case StructTelemetryField::Pose:
757 stt.Set(drive->GetPose());
758 break;
759 default:
760 break;
761 }
762 }
763 for (auto& [field, stt] : m_config.GetChassisSpeedsFields()) {
764 if (!stt.IsEnabled()) continue;
765 switch (field) {
766 case StructTelemetryField::DesiredRobotRelativeChassisSpeeds:
767 stt.Set(drive->GetDesiredChassisSpeeds());
768 break;
769 case StructTelemetryField::CurrentRobotRelativeChassisSpeeds:
770 stt.Set(drive->GetRobotRelativeSpeed());
771 break;
772 case StructTelemetryField::FieldRelativeChassisSpeeds:
773 stt.Set(drive->GetFieldRelativeSpeed());
774 break;
775 default:
776 break;
777 }
778 }
779 for (auto& [field, dt] : m_config.GetDoubleFields()) {
780 if (!dt.IsEnabled()) continue;
781 switch (field) {
782 case DoubleTelemetryField::Gyro:
783 dt.Set(drive->GetGyroAngle().value());
784 break;
785 default:
786 break;
787 }
788 }
789 for (auto& [field, stat] : m_config.GetModuleStatesFields()) {
790 if (!stat.IsEnabled()) continue;
791 switch (field) {
792 case StructArrayTelemetryField::DesiredModuleStates: {
793 auto states = drive->GetDesiredModuleStates();
794 stat.Set(std::vector<frc::SwerveModuleState>(states.begin(), states.end()));
795 break;
796 }
797 case StructArrayTelemetryField::CurrentModuleStates: {
798 auto states = drive->GetModuleStates();
799 stat.Set(std::vector<frc::SwerveModuleState>(states.begin(), states.end()));
800 break;
801 }
802 default:
803 break;
804 }
805 }

808template <std::size_t NumModules>
809void SwerveDriveTelemetry::ApplyTuningValues(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
810 bool nt4Enabled = m_config.GetNT4Enabled();
811
812 auto& boolFields = m_config.GetBoolFields();
813 auto& autoAlignBt = boolFields.at(BooleanTelemetryField::AutoAlignEnabled);
814 auto& driveTuningBt = boolFields.at(BooleanTelemetryField::ModulesDriveTuningEnabled);
815 auto& azimuthTuningBt = boolFields.at(BooleanTelemetryField::ModulesAzimuthTuningEnabled);
816 auto& driveInPlaceBt = boolFields.at(BooleanTelemetryField::ModulesDriveInPlace);
817
818 bool autoAlignOn = autoAlignBt.IsEnabled() && nt4Enabled && autoAlignBt.Get();
819 bool driveTuningOn = driveTuningBt.IsEnabled() && nt4Enabled && driveTuningBt.Get();
820 bool azimuthTuningOn = azimuthTuningBt.IsEnabled() && nt4Enabled && azimuthTuningBt.Get();
821 bool driveInPlaceOn = driveInPlaceBt.IsEnabled() && nt4Enabled && driveInPlaceBt.Get();
822
823 // Only one tuning mode may drive the chassis at a time; enforce priority order
824 // (auto-align > drive tuning > azimuth tuning) and force the losers back off in
825 // NetworkTables so the dashboard reflects what's actually happening.
826 if (autoAlignOn) {
827 if (driveTuningOn) {
828 driveTuningBt.ForceSet(false);
829 driveTuningOn = false;
830 }
831 if (azimuthTuningOn) {
832 azimuthTuningBt.ForceSet(false);
833 azimuthTuningOn = false;
834 }
835 } else if (driveTuningOn && azimuthTuningOn) {
836 azimuthTuningBt.ForceSet(false);
837 azimuthTuningOn = false;
838 }
839
840 if (autoAlignOn) {
841 auto& doubleFields = m_config.GetDoubleFields();
842 double x = doubleFields.at(DoubleTelemetryField::AutoAlignPoseX).Get();
843 double y = doubleFields.at(DoubleTelemetryField::AutoAlignPoseY).Get();
844 double rot = doubleFields.at(DoubleTelemetryField::AutoAlignPoseRotation).Get();
845 frc::Pose2d targetPose{units::meter_t{x}, units::meter_t{y},
846 frc::Rotation2d{units::degree_t{rot}}};
847 drive->SetRobotRelativeChassisSpeeds(drive->DriveToPoseSetpoint(targetPose));
848 }
849
850 if (driveTuningBt.IsEnabled() && nt4Enabled) {
851 auto& modules = drive->GetConfig().GetModules();
852 if (driveTuningOn) {
853 units::meters_per_second_t velocity{
854 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesDriveVelocity).Get()};
855 wpi::array<frc::SwerveModuleState, NumModules> states{wpi::empty_array};
856 for (size_t i = 0; i < NumModules; ++i) {
857 if (driveInPlaceOn) {
858 // Point each module tangent to its position around the robot center so a positive
859 // velocity spins the robot counter-clockwise (WPILib's positive rotation direction).
860 auto moduleLocation = *modules[i]->GetConfig().GetLocation();
861 auto tangentAngle = moduleLocation.Angle() + frc::Rotation2d{units::degree_t{90}};
862 states[i] = frc::SwerveModuleState{velocity, tangentAngle};
863 } else {
864 states[i] = frc::SwerveModuleState{velocity, frc::Rotation2d{}};
865 }
866 }
867 drive->SetSwerveModuleStates(states);
868 } else if (!autoAlignOn) {
869 // Drive tuning is off (and auto-align isn't driving the chassis instead); make sure the
870 // modules don't keep spinning at whatever velocity was last commanded while it was on.
871 wpi::array<frc::SwerveModuleState, NumModules> states{wpi::empty_array};
872 for (size_t i = 0; i < NumModules; ++i) {
873 states[i] =
874 frc::SwerveModuleState{units::meters_per_second_t{0}, modules[i]->GetState().angle};
875 }
876 drive->SetSwerveModuleStates(states);
877 }
878 }
879
880 if (azimuthTuningOn) {
881 frc::Rotation2d angle{units::degree_t{
882 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesAzimuthAngle).Get()}};
883 wpi::array<frc::SwerveModuleState, NumModules> states{wpi::empty_array};
884 for (size_t i = 0; i < NumModules; ++i) {
885 states[i] = frc::SwerveModuleState{units::meters_per_second_t{0}, angle};
886 }
887 drive->SetSwerveModuleStates(states);
888 }
889
890 auto translationPID = drive->GetConfig().GetTranslationPID();
891 auto rotationPID = drive->GetConfig().GetRotationPID();
892 for (auto& [field, dt] : m_config.GetDoubleFields()) {
893 if (!dt.IsTunable()) continue;
894 switch (field) {
895 case DoubleTelemetryField::TranslationP:
896 translationPID.SetP(dt.Get());
897 drive->SetTranslationPID(translationPID);
898 break;
899 case DoubleTelemetryField::TranslationI:
900 translationPID.SetI(dt.Get());
901 drive->SetTranslationPID(translationPID);
902 break;
903 case DoubleTelemetryField::TranslationD:
904 translationPID.SetD(dt.Get());
905 drive->SetTranslationPID(translationPID);
906 break;
907 case DoubleTelemetryField::RotationP:
908 rotationPID.SetP(dt.Get());
909 drive->SetRotationPID(rotationPID);
910 break;
911 case DoubleTelemetryField::RotationI:
912 rotationPID.SetI(dt.Get());
913 drive->SetRotationPID(rotationPID);
914 break;
915 case DoubleTelemetryField::RotationD:
916 rotationPID.SetD(dt.Get());
917 drive->SetRotationPID(rotationPID);
918 break;
919 case DoubleTelemetryField::ModulesDriveP:
920 for (auto* module : drive->GetConfig().GetModules()) {
921 module->GetDriveMotorController()->SetKp(dt.Get());
922 }
923 break;
924 case DoubleTelemetryField::ModulesDriveI:
925 for (auto* module : drive->GetConfig().GetModules()) {
926 module->GetDriveMotorController()->SetKi(dt.Get());
927 }
928 break;
929 case DoubleTelemetryField::ModulesDriveD:
930 for (auto* module : drive->GetConfig().GetModules()) {
931 module->GetDriveMotorController()->SetKd(dt.Get());
932 }
933 break;
934 case DoubleTelemetryField::ModulesAzimuthP:
935 for (auto* module : drive->GetConfig().GetModules()) {
936 module->GetAzimuthMotorController()->SetKp(dt.Get());
937 }
938 break;
939 case DoubleTelemetryField::ModulesAzimuthI:
940 for (auto* module : drive->GetConfig().GetModules()) {
941 module->GetAzimuthMotorController()->SetKi(dt.Get());
942 }
943 break;
944 case DoubleTelemetryField::ModulesAzimuthD:
945 for (auto* module : drive->GetConfig().GetModules()) {
946 module->GetAzimuthMotorController()->SetKd(dt.Get());
947 }
948 break;
949 case DoubleTelemetryField::ModulesDriveKs:
950 for (auto* module : drive->GetConfig().GetModules()) {
951 module->GetDriveMotorController()->SetKs(dt.Get());
952 }
953 break;
954 case DoubleTelemetryField::ModulesDriveKv:
955 for (auto* module : drive->GetConfig().GetModules()) {
956 module->GetDriveMotorController()->SetKv(dt.Get());
957 }
958 break;
959 case DoubleTelemetryField::ModulesDriveKa:
960 for (auto* module : drive->GetConfig().GetModules()) {
961 module->GetDriveMotorController()->SetKa(dt.Get());
962 }
963 break;
964 case DoubleTelemetryField::ModulesAzimuthKs:
965 for (auto* module : drive->GetConfig().GetModules()) {
966 module->GetAzimuthMotorController()->SetKs(dt.Get());
967 }
968 break;
969 case DoubleTelemetryField::ModulesAzimuthKv:
970 for (auto* module : drive->GetConfig().GetModules()) {
971 module->GetAzimuthMotorController()->SetKv(dt.Get());
972 }
973 break;
974 case DoubleTelemetryField::ModulesAzimuthKa:
975 for (auto* module : drive->GetConfig().GetModules()) {
976 module->GetAzimuthMotorController()->SetKa(dt.Get());
977 }
978 break;
979 default:
980 break;
981 }
982 }
