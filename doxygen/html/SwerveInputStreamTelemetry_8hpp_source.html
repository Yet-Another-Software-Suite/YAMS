192 for (auto& [field, dt] : m_doubleFields) {

213 // Publish the stream's values to the Tuning table.
214 ApplyTuningValues();
215 // No requirements, so tuning does not interrupt the command driving with the stream.
216 m_liveTuningCommand.emplace(
217 wpi::cmd::Run([this] { ApplyTuningValues(); }).WithName("Live Tuning"));
218 telemetry::EnsureTuningTunableBackend();
219 wpi::tunables::Publish(m_liveTuningPath, *m_liveTuningCommand->get());

248 void UpdateTelemetry() {
249 for (auto& [field, st] : m_stringFields) {
250 if (!st.IsEnabled()) continue;
251 switch (field) {
252 case StringTelemetryField::Mode:
253 st.Set(m_stream->GetCurrentModeName());
254 break;
255 }
256 }
257 for (auto& [field, dt] : m_doubleFields) {
258 if (!dt.IsEnabled()) continue;
259 switch (field) {
260 case DoubleTelemetryField::Deadband:
261 dt.Set(m_stream->GetAxisDeadband());
262 break;
263 case DoubleTelemetryField::TranslationScale:
264 dt.Set(m_stream->GetTranslationAxisScale());
265 break;
266 case DoubleTelemetryField::RotationScale:
267 dt.Set(m_stream->GetOmegaAxisScale());
268 break;
269 case DoubleTelemetryField::MaxLinearVelocity:
270 dt.Set(m_stream->GetMaximumChassisLinearVelocity().value());
271 break;
272 case DoubleTelemetryField::MaxAngularVelocity:
273 dt.Set(m_stream->GetMaximumChassisAngularVelocity().value());
274 break;
275 case DoubleTelemetryField::RotationControllerMeasurement: {
276 auto measurement = m_stream->GetRotationControllerMeasurement();
277 dt.Set(measurement ? measurement->value() : std::numeric_limits<double>::quiet_NaN());
278 break;
279 }
280 case DoubleTelemetryField::RotationControllerSetpoint: {
281 auto setpoint = m_stream->GetRotationControllerSetpoint();
282 dt.Set(setpoint ? setpoint->value() : std::numeric_limits<double>::quiet_NaN());
283 break;
284 }
285 default:
286 // Tunable fields are published by ApplyTuningValues.
287 break;
288 }
289 }
290 for (auto& [field, bt] : m_boolFields) {
291 if (!bt.IsEnabled()) continue;
292 switch (field) {
293 case BooleanTelemetryField::TranslationCube:
294 bt.Set(m_stream->IsTranslationCubeEnabled());
295 break;
296 case BooleanTelemetryField::RotationCube:
297 bt.Set(m_stream->IsOmegaCubeEnabled());
298 break;
299 case BooleanTelemetryField::AllianceRelative:
300 bt.Set(m_stream->IsAllianceRelativeEnabled());
301 break;
302 case BooleanTelemetryField::RobotRelative:
303 bt.Set(m_stream->IsRobotRelativeEnabled());
304 break;
305 default:
306 // Tunable fields are published by ApplyTuningValues.
307 break;
308 }
309 }
310 }

316 void ApplyTuningValues() {
317 for (auto& [field, dt] : m_doubleFields) {
318 if (!dt.IsEnabled()) continue;
319 switch (field) {
320 case DoubleTelemetryField::TunableDeadband:
321 if (dt.IsTunable() && dt.Get() >= 0.0 && dt.Get() < 1.0) {
322 m_stream->SetAxisDeadband(dt.Get());
323 }
324 PublishStreamValue(dt, m_stream->GetAxisDeadband());
325 break;
326 case DoubleTelemetryField::TunableTranslationScale:
327 if (dt.IsTunable() && dt.Get() > 0.0 && dt.Get() <= 1.0) {
328 m_stream->SetTranslationAxisScale(dt.Get());
329 }
330 PublishStreamValue(dt, m_stream->GetTranslationAxisScale());
331 break;
332 case DoubleTelemetryField::TunableRotationScale:
333 if (dt.IsTunable() && dt.Get() > 0.0 && dt.Get() <= 1.0) {
334 m_stream->SetOmegaAxisScale(dt.Get());
335 }
336 PublishStreamValue(dt, m_stream->GetOmegaAxisScale());
337 break;
338 case DoubleTelemetryField::TunableMaxLinearVelocity:
339 if (dt.IsTunable() && dt.Get() > 0.0 && std::isfinite(dt.Get())) {
340 m_stream->SetMaximumChassisLinearVelocity(wpi::units::meters_per_second_t{dt.Get()});
341 }
342 PublishStreamValue(dt, m_stream->GetMaximumChassisLinearVelocity().value());
343 break;
344 case DoubleTelemetryField::TunableMaxAngularVelocity:
345 if (dt.IsTunable() && dt.Get() > 0.0 && std::isfinite(dt.Get())) {
346 m_stream->SetMaximumChassisAngularVelocity(wpi::units::radians_per_second_t{dt.Get()});
347 }
348 PublishStreamValue(dt, m_stream->GetMaximumChassisAngularVelocity().value());
349 break;
350 default:
351 // Read-only fields are published by UpdateTelemetry.
352 break;
353 }
354 }
355 for (auto& [field, bt] : m_boolFields) {
356 if (!bt.IsEnabled()) continue;
357 switch (field) {
358 case BooleanTelemetryField::TunableTranslationCube:
359 if (bt.IsTunable()) {
360 m_stream->SetTranslationCubeEnabled(bt.Get());
361 }
362 PublishStreamValue(bt, m_stream->IsTranslationCubeEnabled());
363 break;
364 case BooleanTelemetryField::TunableRotationCube:
365 if (bt.IsTunable()) {
366 m_stream->SetOmegaCubeEnabled(bt.Get());
367 }
368 PublishStreamValue(bt, m_stream->IsOmegaCubeEnabled());
369 break;
370 case BooleanTelemetryField::TunableAllianceRelative:
371 if (bt.IsTunable()) {
372 m_stream->SetAllianceRelativeEnabled(bt.Get());
373 }
374 PublishStreamValue(bt, m_stream->IsAllianceRelativeEnabled());
375 break;
376 case BooleanTelemetryField::TunableRobotRelative:
377 if (bt.IsTunable()) {
378 m_stream->SetRobotRelativeEnabled(bt.Get());
379 }
380 PublishStreamValue(bt, m_stream->IsRobotRelativeEnabled());
381 break;
382 default:
383 // Read-only fields are published by UpdateTelemetry.
384 break;
385 }
386 }

419 template <typename Telemetry, typename T>
420 static void PublishStreamValue(Telemetry& telemetry, T streamValue) {
421 if (telemetry.Get() != streamValue) {
422 telemetry.ForceSet(streamValue);
423 telemetry.IsTunable();
424 }
