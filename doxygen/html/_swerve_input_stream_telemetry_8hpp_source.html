63 SwerveInputStreamTelemetry(SwerveInputStream<NumModules>& stream, std::string_view name,
64 TelemetryVerbosity verbosity)
65 : m_stream{&stream},
66 m_liveTuningPath{"Mechanisms/Commands/SwerveInputStream/" + std::string{name} +
67 "/Live Tuning"} {
68 auto instance = nt::NetworkTableInstance::GetDefault();
69 auto dataTable = instance.GetTable("SwerveInputStream")->GetSubTable(name);
70 m_modePublisher = dataTable->GetStringTopic("mode").Publish();
71
72 if (verbosity == TelemetryVerbosity::MEDIUM || verbosity == TelemetryVerbosity::HIGH) {
73 PublishConfig(*dataTable);
74 }
75 if (verbosity == TelemetryVerbosity::HIGH) {
76 AddTunableValues(
77 *instance.GetTable("Tuning")->GetSubTable("SwerveInputStream")->GetSubTable(name));
78 // No requirements, so tuning does not interrupt the command driving with the stream.
79 m_liveTuningCommand.emplace(
80 frc2::cmd::Run([this] { ApplyTuningValues(); }).WithName("Live Tuning"));
81 frc::SmartDashboard::PutData(m_liveTuningPath, m_liveTuningCommand->get());
82 }
83 UpdateTelemetry();
84 }

135 template <typename T, typename Entry>
136 class TunableValue {
137 public:
147 TunableValue(nt::NetworkTable& table, std::string_view key, std::function<T()> getter,
148 std::function<void(T)> setter,
149 std::function<bool(T)> valid = [](T) { return true; })
150 : m_getter{std::move(getter)},
151 m_setter{std::move(setter)},
152 m_valid{std::move(valid)},
153 m_lastValue{m_getter()},
154 m_entry{MakeEntry(table, key, m_lastValue)} {
155 m_entry.Set(m_lastValue);
156 }
157
159 void Update() {
160 T published = m_entry.Get();
161 if (published != m_lastValue && m_valid(published)) {
162 m_setter(published);
163 }
164 T current = m_getter();
165 if (current != published) {
166 m_entry.Set(current);
167 }
168 m_lastValue = current;
169 }
170
171 private:
172 static Entry MakeEntry(nt::NetworkTable& table, std::string_view key, T defaultValue) {
173 if constexpr (std::is_same_v<T, bool>) {
174 return table.GetBooleanTopic(key).GetEntry(defaultValue);
175 } else {
176 return table.GetDoubleTopic(key).GetEntry(defaultValue);
177 }
178 }
179
180 std::function<T()> m_getter;
181 std::function<void(T)> m_setter;
182 std::function<bool(T)> m_valid;
184 T m_lastValue;
185 Entry m_entry;
186 };

188 void PublishConfig(nt::NetworkTable& table) {
189 PublishDouble(table, "deadband", [this] { return m_stream->GetAxisDeadband(); });
190 PublishDouble(table, "translationScale", [this] { return m_stream->GetTranslationAxisScale(); });
191 PublishDouble(table, "rotationScale", [this] { return m_stream->GetOmegaAxisScale(); });
192 PublishDouble(table, "maxLinearVelocity",
193 [this] { return m_stream->GetMaximumChassisLinearVelocity().value(); });
194 PublishDouble(table, "maxAngularVelocity",
195 [this] { return m_stream->GetMaximumChassisAngularVelocity().value(); });
196 PublishBoolean(table, "translationCube", [this] { return m_stream->IsTranslationCubeEnabled(); });
197 PublishBoolean(table, "rotationCube", [this] { return m_stream->IsOmegaCubeEnabled(); });
198 PublishBoolean(table, "allianceRelative",
199 [this] { return m_stream->IsAllianceRelativeEnabled(); });
200 PublishBoolean(table, "robotRelative", [this] { return m_stream->IsRobotRelativeEnabled(); });

219 void AddTunableValues(nt::NetworkTable& table) {
220 m_doubles.emplace_back(
221 table, "deadband", [this] { return m_stream->GetAxisDeadband(); },
222 [this](double value) { m_stream->SetAxisDeadband(value); },
223 [](double value) { return value >= 0.0 && value < 1.0; });
224 m_doubles.emplace_back(
225 table, "translationScale", [this] { return m_stream->GetTranslationAxisScale(); },
226 [this](double value) { m_stream->SetTranslationAxisScale(value); },
227 [](double value) { return value > 0.0 && value <= 1.0; });
228 m_doubles.emplace_back(
229 table, "rotationScale", [this] { return m_stream->GetOmegaAxisScale(); },
230 [this](double value) { m_stream->SetOmegaAxisScale(value); },
231 [](double value) { return value > 0.0 && value <= 1.0; });
232 m_doubles.emplace_back(
233 table, "maxLinearVelocity",
234 [this] { return m_stream->GetMaximumChassisLinearVelocity().value(); },
235 [this](double value) {
236 m_stream->SetMaximumChassisLinearVelocity(units::meters_per_second_t{value});
237 },
238 [](double value) { return value > 0.0 && std::isfinite(value); });
239 m_doubles.emplace_back(
240 table, "maxAngularVelocity",
241 [this] { return m_stream->GetMaximumChassisAngularVelocity().value(); },
242 [this](double value) {
243 m_stream->SetMaximumChassisAngularVelocity(units::radians_per_second_t{value});
244 },
245 [](double value) { return value > 0.0 && std::isfinite(value); });
246 m_booleans.emplace_back(
247 table, "translationCube", [this] { return m_stream->IsTranslationCubeEnabled(); },
248 [this](bool value) { m_stream->SetTranslationCubeEnabled(value); });
249 m_booleans.emplace_back(
250 table, "rotationCube", [this] { return m_stream->IsOmegaCubeEnabled(); },
251 [this](bool value) { m_stream->SetOmegaCubeEnabled(value); });
252 m_booleans.emplace_back(
253 table, "allianceRelative", [this] { return m_stream->IsAllianceRelativeEnabled(); },
254 [this](bool value) { m_stream->SetAllianceRelativeEnabled(value); });
255 m_booleans.emplace_back(
256 table, "robotRelative", [this] { return m_stream->IsRobotRelativeEnabled(); },
257 [this](bool value) { m_stream->SetRobotRelativeEnabled(value); });
258 }
