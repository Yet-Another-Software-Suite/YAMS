189 frc2::CommandPtr DriveToPose(frc::Pose2d pose) {
190 return frc2::cmd::RunOnce([this] {
191 ResetTranslationPID();
192 ResetAzimuthPID();
193 })
194 .AndThen(Drive([this, pose]() -> frc::ChassisSpeeds {
195 auto& azimuthPID = m_config->GetRotationPID();
196 auto& translationPID = m_config->GetTranslationPID();
197 auto currentPose = GetPose();
198 auto distance = GetDistanceFromPose(pose);
199 auto translationScalar = translationPID.Calculate(distance.value(), 0.0);
200 auto poseDiff = currentPose.RelativeTo(pose);
201 return frc::ChassisSpeeds::FromFieldRelativeSpeeds(
202 frc::ChassisSpeeds{
203 units::meters_per_second_t{poseDiff.X().value() * translationScalar},
204 units::meters_per_second_t{poseDiff.Y().value() * translationScalar},
205 units::radians_per_second_t{
206 azimuthPID.Calculate(currentPose.Rotation().Radians().value(),
207 pose.Rotation().Radians().value())}},
208 frc::Rotation2d{units::radian_t{GetGyroAngle()}});
209 }))
210 .WithName("Drive to Pose");
211 }

332 void UpdateTelemetry() {
333 UpdatePoseEstimator();
334 auto pose = GetPose();
335 auto currentStates = GetModuleStates();
336
337 m_gyroPublisher(units::degree_t{GetGyroAngle()}.value());
338 m_desiredModuleStatesPublisher(m_desiredModuleStates);
339 m_currentModuleStatesPublisher(currentStates);
340 m_posePublisher(pose);
341 m_desiredRobotRelChassisSpeedsPublisher(m_desiredChassisSpeeds);
342 m_currentRobotRelChassisSpeedsPublisher(GetRobotRelativeSpeed());
343 m_fieldRelChassisSpeedsPublisher(GetFieldRelativeSpeed());
344
345 for (auto* mod : m_config->GetModules()) {
346 mod->UpdateTelemetry();
347 }
348
349 m_field2d.SetRobotPose(pose);
350 std::vector<frc::Pose2d> modulePoses;
351 modulePoses.reserve(NumModules);
352 for (size_t i = 0; i < NumModules; ++i) {
353 auto* mod = m_config->GetModules()[i];
354 auto location = *mod->GetConfig().GetLocation();
355 auto moduleTranslation = pose.Translation() + location.RotateBy(pose.Rotation());
356 auto moduleHeading = pose.Rotation() + currentStates[i].angle;
357 modulePoses.push_back(frc::Pose2d{moduleTranslation, moduleHeading});
358 }
359 m_field2d.GetObject("modules")->SetPoses(modulePoses);
360 }
361
368 void SimIterate() {
369 if (!m_simTimer.IsRunning()) m_simTimer.Start();
370 for (auto* mod : m_config->GetModules()) {
371 mod->SimIterate();
372 }
373 auto speeds = m_kinematics.ToChassisSpeeds(GetModuleStates());
374 m_simGyroAngle += units::degree_t{units::degrees_per_second_t{speeds.omega}.value() *
375 m_simTimer.Get().value()};
376 m_simTimer.Reset();
377 }

499 static frc::SwerveDriveKinematics<NumModules> BuildKinematics(const SwerveDriveConfig& config) {
500 assert(config.GetModules().size() == NumModules);
501 wpi::array<frc::Translation2d, NumModules> locations{wpi::empty_array};
502 for (size_t i = 0; i < NumModules; ++i) {
