204 wpi::math::ChassisVelocities DriveToPoseSetpoint(wpi::math::Pose2d targetPose) {
205 auto rotationPIDRef = m_config->GetRotationPID();
206 if (!rotationPIDRef) {
207 throw exceptions::SwerveDriveConfigurationException(
208 "No rotation PID controller configured, cannot drive to pose. Use "
209 "SwerveDriveConfig::WithRotationController(PIDController) to fix this error.");
210 }
211 auto translationPIDRef = m_config->GetTranslationPID();
212 if (!translationPIDRef) {
213 throw exceptions::SwerveDriveConfigurationException(
214 "No translation PID controller configured, cannot drive to pose. Use "
215 "SwerveDriveConfig::WithTranslationController(PIDController) to fix this error.");
216 }
217 auto& rotationPID = rotationPIDRef->get();
218 auto& translationPID = translationPIDRef->get();
219 auto distance = GetDistanceFromPose(targetPose);
220 auto translationScalar = translationPID.Calculate(distance.value(), 0.0);
221 auto currentPose = GetPose();
222 // Plain field-frame translation delta (not RelativeTo/operator-, which expresses the result
223 // in targetPose's rotated frame and would skew the commanded direction whenever targetPose's
224 // heading is non-zero).
225 auto translationDiff = currentPose.Translation() - targetPose.Translation();
226 return (wpi::math::ChassisVelocities{
227 wpi::units::meters_per_second_t{translationDiff.X().value() * translationScalar},
228 wpi::units::meters_per_second_t{translationDiff.Y().value() * translationScalar},
229 wpi::units::radians_per_second_t{
230 rotationPID.Calculate(currentPose.Rotation().Radians().value(),
231 targetPose.Rotation().Radians().value())}})
232 .ToRobotRelative(wpi::math::Rotation2d{wpi::units::radian_t{GetGyroAngle()}});
233 }

371 void UpdateTelemetry() {
372 UpdatePoseEstimator();
373 auto pose = GetPose();
374 auto currentStates = GetModuleStates();
375
376 m_swerveTelemetry->Publish(this);
377
378 for (auto* mod : m_config->GetModules()) {
379 mod->UpdateTelemetry();
380 }
381 m_telemetry.UpdateLoopTime();
382
383 m_field2d.SetRobotPose(pose);
384 std::vector<wpi::math::Pose2d> modulePoses;
385 modulePoses.reserve(NumModules);
386 for (size_t i = 0; i < NumModules; ++i) {
387 auto* mod = m_config->GetModules()[i];
388 auto location = *mod->GetConfig().GetLocation();
389 auto moduleTranslation = pose.Translation() + location.RotateBy(pose.Rotation());
390 auto moduleHeading = pose.Rotation() + currentStates[i].angle;
391 modulePoses.push_back(wpi::math::Pose2d{moduleTranslation, moduleHeading});
392 }
393 m_field2d.GetObject("modules")->SetPoses(modulePoses);
394 }

403 void SimIterate() {
404 if (!m_simTimer.IsRunning()) m_simTimer.Start();
405 for (auto* mod : m_config->GetModules()) {
406 mod->SimIterate();
407 }
408 auto desired = m_kinematics.ToChassisVelocities(m_desiredModuleStates);
409
410 auto dt = m_simTimer.Get();
411 wpi::math::Twist2d twist{desired.vx * dt, desired.vy * dt, desired.omega * dt};
412 m_simPose = m_simPose + twist.Exp();
413
414 auto speeds = m_kinematics.ToChassisVelocities(GetModuleStates());
415 m_simGyroAngle +=
416 wpi::units::degree_t{wpi::units::degrees_per_second_t{speeds.omega}.value() * dt.value()};
417 m_simTimer.Reset();
418 }

613 void SetupTelemetry() {
614 using TelemetryVerbosity = motorcontrollers::SmartMotorControllerConfig::TelemetryVerbosity;
615 telemetry::SwerveDriveTelemetryConfig telemetryCfg;
616 if (auto specified = m_config->GetSwerveDriveTelemetryConfig()) {
617 telemetryCfg = std::move(*specified);
618 } else {
619 telemetryCfg.WithTelemetryVerbosity(
620 m_config->GetTelemetryVerbosity().value_or(TelemetryVerbosity::HIGH));
621 }
622
623 if (auto dataLogName = telemetryCfg.GetDataLogName()) {
624 m_telemetry.SetupTelemetry(GetName(), *dataLogName);
625 } else {
626 m_telemetry.SetupTelemetry(GetName());
627 }
628
629 m_swerveTelemetry.emplace(std::move(telemetryCfg));
630 m_swerveTelemetry->SetupTelemetry(this);
631
632 // Drive to pose tuning needs both controllers, so only offer it when they are configured.
633 if (m_config->GetTranslationPID() && m_config->GetRotationPID()) {
634 m_driveToPoseTuningCommand.emplace(wpi::cmd::StartRun(
635 [this] {
636 std::puts(
637 "================= Starting SwerveDrive.driveToPoseTuning() =================\n");
638 ResetTranslationPID();
639 ResetRotationPID();
640 },
641 [this] { m_swerveTelemetry->ApplyTuningValues(this); }));
642 telemetry::EnsureMechanismsTunableBackend();
643 wpi::tunables::Publish("Mechanisms/" + GetName() + "/tuning/driveToPose",
644 *m_driveToPoseTuningCommand->get());
645 }
646 }

695// SwerveDrive's module count so that SwerveDriveTelemetry itself need not be a template. Their

697// only place both SwerveDriveTelemetry's private members and SwerveDrive<NumModules>'s interface
698// are simultaneously visible.

700template <std::size_t NumModules>
701void SwerveDriveTelemetry::SetupTelemetry(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
702 auto driveName = drive->GetName();
703 for (auto* module : drive->GetConfig().GetModules()) {
704 module->SetupTelemetry(driveName);
705 }
706
707 auto inst = wpi::nt::NetworkTableInstance::GetDefault();
708 m_dataTable = inst.GetTable("Mechanisms")->GetSubTable(driveName);
709 m_tuningTable = inst.GetTable("Tuning")->GetSubTable(driveName);
710
711 bool nt4Enabled = m_config.GetNT4Enabled();
712 auto dataLogName = m_config.GetDataLogName();
713
714 {
715 // Only publish the drive to pose gains and auto-align fields for controllers that exist.
716 auto translationPID = drive->GetConfig().GetTranslationPID();
717 auto rotationPID = drive->GetConfig().GetRotationPID();
718 auto& doubleFields = m_config.GetDoubleFields();
719 if (translationPID) {
720 doubleFields.at(DoubleTelemetryField::TranslationP)
721 .SetDefaultValue(translationPID->get().GetP());
722 doubleFields.at(DoubleTelemetryField::TranslationI)
723 .SetDefaultValue(translationPID->get().GetI());
724 doubleFields.at(DoubleTelemetryField::TranslationD)
725 .SetDefaultValue(translationPID->get().GetD());
726 } else {
727 doubleFields.at(DoubleTelemetryField::TranslationP).Disable();
728 doubleFields.at(DoubleTelemetryField::TranslationI).Disable();
729 doubleFields.at(DoubleTelemetryField::TranslationD).Disable();
730 }
731 if (rotationPID) {
732 doubleFields.at(DoubleTelemetryField::RotationP).SetDefaultValue(rotationPID->get().GetP());
733 doubleFields.at(DoubleTelemetryField::RotationI).SetDefaultValue(rotationPID->get().GetI());
734 doubleFields.at(DoubleTelemetryField::RotationD).SetDefaultValue(rotationPID->get().GetD());
735 } else {
736 doubleFields.at(DoubleTelemetryField::RotationP).Disable();
737 doubleFields.at(DoubleTelemetryField::RotationI).Disable();
738 doubleFields.at(DoubleTelemetryField::RotationD).Disable();
739 }
740 // Auto-align drives to a pose, which needs both controllers.
741 if (!translationPID || !rotationPID) {
742 doubleFields.at(DoubleTelemetryField::AutoAlignPoseX).Disable();
743 doubleFields.at(DoubleTelemetryField::AutoAlignPoseY).Disable();
744 doubleFields.at(DoubleTelemetryField::AutoAlignPoseRotation).Disable();
745 m_config.GetBoolFields().at(BooleanTelemetryField::AutoAlignEnabled).Disable();
746 }
747
748 auto& modules = drive->GetConfig().GetModules();
749 if (!modules.empty()) {
750 auto* driveMotor = modules[0]->GetDriveMotorController();
751 auto driveGains =
752 driveMotor->GetConfig().GetSlotGains(driveMotor->GetClosedLoopControllerSlot());
753 m_config.GetDoubleFields()
754 .at(DoubleTelemetryField::ModulesDriveP)
755 .SetDefaultValue(driveGains.kP);
756 m_config.GetDoubleFields()
757 .at(DoubleTelemetryField::ModulesDriveI)
758 .SetDefaultValue(driveGains.kI);
759 m_config.GetDoubleFields()
760 .at(DoubleTelemetryField::ModulesDriveD)
761 .SetDefaultValue(driveGains.kD);
762 m_config.GetDoubleFields()
763 .at(DoubleTelemetryField::ModulesDriveKs)
764 .SetDefaultValue(driveGains.kS);
765 m_config.GetDoubleFields()
766 .at(DoubleTelemetryField::ModulesDriveKv)
767 .SetDefaultValue(driveGains.kV);
768 m_config.GetDoubleFields()
769 .at(DoubleTelemetryField::ModulesDriveKa)
770 .SetDefaultValue(driveGains.kA);
771
772 auto* azimuthMotor = modules[0]->GetAzimuthMotorController();
773 auto azimuthGains =
774 azimuthMotor->GetConfig().GetSlotGains(azimuthMotor->GetClosedLoopControllerSlot());
775 m_config.GetDoubleFields()
776 .at(DoubleTelemetryField::ModulesAzimuthP)
777 .SetDefaultValue(azimuthGains.kP);
778 m_config.GetDoubleFields()
779 .at(DoubleTelemetryField::ModulesAzimuthI)
780 .SetDefaultValue(azimuthGains.kI);
781 m_config.GetDoubleFields()
782 .at(DoubleTelemetryField::ModulesAzimuthD)
783 .SetDefaultValue(azimuthGains.kD);
784 m_config.GetDoubleFields()
785 .at(DoubleTelemetryField::ModulesAzimuthKs)
786 .SetDefaultValue(azimuthGains.kS);
787 m_config.GetDoubleFields()
788 .at(DoubleTelemetryField::ModulesAzimuthKv)
789 .SetDefaultValue(azimuthGains.kV);
790 m_config.GetDoubleFields()
791 .at(DoubleTelemetryField::ModulesAzimuthKa)
792 .SetDefaultValue(azimuthGains.kA);
793 }
794 }
795
796 for (auto& [field, dt] : m_config.GetDoubleFields()) {
797 if (!dt.IsEnabled()) continue;
798 if (nt4Enabled) dt.SetupNetworkTables(m_dataTable, m_tuningTable);
799 if (dataLogName) dt.SetupDataLog(*dataLogName);
800 }
801 for (auto& [field, bt] : m_config.GetBoolFields()) {
802 if (!bt.IsEnabled()) continue;
803 if (nt4Enabled) bt.SetupNetworkTables(m_dataTable, m_tuningTable);
804 if (dataLogName) bt.SetupDataLog(*dataLogName);
805 }
806 for (auto& [field, stt] : m_config.GetPoseFields()) {
807 if (!stt.IsEnabled()) continue;
808 if (nt4Enabled) stt.SetupNetworkTables(m_dataTable, m_tuningTable);
809 if (dataLogName) stt.SetupDataLog(*dataLogName);
810 }
811 for (auto& [field, stt] : m_config.GetChassisSpeedsFields()) {
812 if (!stt.IsEnabled()) continue;
813 if (nt4Enabled) stt.SetupNetworkTables(m_dataTable, m_tuningTable);
814 if (dataLogName) stt.SetupDataLog(*dataLogName);
815 }
816 for (auto& [field, stat] : m_config.GetModuleStatesFields()) {
817 if (!stat.IsEnabled()) continue;
818 if (nt4Enabled) stat.SetupNetworkTables(m_dataTable, m_tuningTable);
819 if (dataLogName) stat.SetupDataLog(*dataLogName);
820 }

823template <std::size_t NumModules>
824void SwerveDriveTelemetry::Publish(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
825 for (auto& [field, stt] : m_config.GetPoseFields()) {

880template <std::size_t NumModules>
881void SwerveDriveTelemetry::ApplyTuningValues(mechanisms::swerve::SwerveDrive<NumModules>* drive) {
882 bool nt4Enabled = m_config.GetNT4Enabled();
883
884 auto& boolFields = m_config.GetBoolFields();
885 auto& autoAlignBt = boolFields.at(BooleanTelemetryField::AutoAlignEnabled);
886 auto& driveTuningBt = boolFields.at(BooleanTelemetryField::ModulesDriveTuningEnabled);
887 auto& azimuthTuningBt = boolFields.at(BooleanTelemetryField::ModulesAzimuthTuningEnabled);
888 auto& driveInPlaceBt = boolFields.at(BooleanTelemetryField::ModulesDriveInPlace);
889
890 bool autoAlignOn = autoAlignBt.IsEnabled() && nt4Enabled && autoAlignBt.Get();
891 bool driveTuningOn = driveTuningBt.IsEnabled() && nt4Enabled && driveTuningBt.Get();
892 bool azimuthTuningOn = azimuthTuningBt.IsEnabled() && nt4Enabled && azimuthTuningBt.Get();
893 bool driveInPlaceOn = driveInPlaceBt.IsEnabled() && nt4Enabled && driveInPlaceBt.Get();
894
895 // Only one tuning mode may drive the chassis at a time; enforce priority order
896 // (auto-align > drive tuning > azimuth tuning) and force the losers back off in
897 // NetworkTables so the dashboard reflects what's actually happening.
898 if (autoAlignOn) {
899 if (driveTuningOn) {
900 driveTuningBt.ForceSet(false);
901 driveTuningOn = false;
902 }
903 if (azimuthTuningOn) {
904 azimuthTuningBt.ForceSet(false);
905 azimuthTuningOn = false;
906 }
907 } else if (driveTuningOn && azimuthTuningOn) {
908 azimuthTuningBt.ForceSet(false);
909 azimuthTuningOn = false;
910 }
911
912 if (autoAlignOn) {
913 auto& doubleFields = m_config.GetDoubleFields();
914 double x = doubleFields.at(DoubleTelemetryField::AutoAlignPoseX).Get();
915 double y = doubleFields.at(DoubleTelemetryField::AutoAlignPoseY).Get();
916 double rot = doubleFields.at(DoubleTelemetryField::AutoAlignPoseRotation).Get();
917 wpi::math::Pose2d targetPose{wpi::units::meter_t{x}, wpi::units::meter_t{y},
918 wpi::math::Rotation2d{wpi::units::degree_t{rot}}};
919 drive->SetRobotRelativeChassisSpeeds(drive->DriveToPoseSetpoint(targetPose));
920 }
921
922 if (driveTuningBt.IsEnabled() && nt4Enabled) {
923 auto& modules = drive->GetConfig().GetModules();
924 if (driveTuningOn) {
925 wpi::units::meters_per_second_t velocity{
926 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesDriveVelocity).Get()};
927 wpi::util::array<wpi::math::SwerveModuleVelocity, NumModules> states{wpi::util::empty_array};
928 for (size_t i = 0; i < NumModules; ++i) {
929 if (driveInPlaceOn) {
930 // Point each module tangent to its position around the robot center so a positive
931 // velocity spins the robot counter-clockwise (WPILib's positive rotation direction).
932 auto moduleLocation = *modules[i]->GetConfig().GetLocation();
933 auto tangentAngle = moduleLocation.Angle().value_or(wpi::math::Rotation2d{}) +
934 wpi::math::Rotation2d{wpi::units::degree_t{90}};
935 states[i] = wpi::math::SwerveModuleVelocity{velocity, tangentAngle};
936 } else {
937 states[i] = wpi::math::SwerveModuleVelocity{velocity, wpi::math::Rotation2d{}};
938 }
939 }
940 drive->SetSwerveModuleStates(states);
941 } else if (!autoAlignOn) {
942 // Drive tuning is off (and auto-align isn't driving the chassis instead); make sure the
943 // modules don't keep spinning at whatever velocity was last commanded while it was on.
944 wpi::util::array<wpi::math::SwerveModuleVelocity, NumModules> states{wpi::util::empty_array};
945 for (size_t i = 0; i < NumModules; ++i) {
946 states[i] = wpi::math::SwerveModuleVelocity{wpi::units::meters_per_second_t{0},
947 modules[i]->GetState().angle};
948 }
949 drive->SetSwerveModuleStates(states);
950 }
951 }
952
953 if (azimuthTuningOn) {
954 wpi::math::Rotation2d angle{wpi::units::degree_t{
955 m_config.GetDoubleFields().at(DoubleTelemetryField::ModulesAzimuthAngle).Get()}};
956 wpi::util::array<wpi::math::SwerveModuleVelocity, NumModules> states{wpi::util::empty_array};
957 for (size_t i = 0; i < NumModules; ++i) {
958 states[i] = wpi::math::SwerveModuleVelocity{wpi::units::meters_per_second_t{0}, angle};
959 }
960 drive->SetSwerveModuleStates(states);
961 }
962
963 // Copies of the configured controllers; gain fields for missing controllers are disabled.
964 std::optional<wpi::math::PIDController> translationPID;
965 if (auto pid = drive->GetConfig().GetTranslationPID()) translationPID = pid->get();
966 std::optional<wpi::math::PIDController> rotationPID;
967 if (auto pid = drive->GetConfig().GetRotationPID()) rotationPID = pid->get();
968 for (auto& [field, dt] : m_config.GetDoubleFields()) {
969 if (!dt.IsTunable()) continue;
970 switch (field) {
971 case DoubleTelemetryField::TranslationP:
972 if (translationPID) {
973 translationPID->SetP(dt.Get());
974 drive->SetTranslationPID(*translationPID);
975 }
976 break;
977 case DoubleTelemetryField::TranslationI:
978 if (translationPID) {
979 translationPID->SetI(dt.Get());
980 drive->SetTranslationPID(*translationPID);
981 }
982 break;
983 case DoubleTelemetryField::TranslationD:
984 if (translationPID) {
985 translationPID->SetD(dt.Get());
986 drive->SetTranslationPID(*translationPID);
987 }
988 break;
989 case DoubleTelemetryField::RotationP:
990 if (rotationPID) {
991 rotationPID->SetP(dt.Get());
992 drive->SetRotationPID(*rotationPID);
993 }
994 break;
995 case DoubleTelemetryField::RotationI:
996 if (rotationPID) {
997 rotationPID->SetI(dt.Get());
998 drive->SetRotationPID(*rotationPID);
999 }
1000 break;
1001 case DoubleTelemetryField::RotationD:
1002 if (rotationPID) {
1003 rotationPID->SetD(dt.Get());
1004 drive->SetRotationPID(*rotationPID);
1005 }
1006 break;
1007 case DoubleTelemetryField::ModulesDriveP:
1008 for (auto* module : drive->GetConfig().GetModules()) {
1009 module->GetDriveMotorController()->SetKp(dt.Get());
1010 }
1011 break;
1012 case DoubleTelemetryField::ModulesDriveI:
1013 for (auto* module : drive->GetConfig().GetModules()) {
1014 module->GetDriveMotorController()->SetKi(dt.Get());
1015 }
1016 break;
1017 case DoubleTelemetryField::ModulesDriveD:
1018 for (auto* module : drive->GetConfig().GetModules()) {
1019 module->GetDriveMotorController()->SetKd(dt.Get());
1020 }
1021 break;
1022 case DoubleTelemetryField::ModulesAzimuthP:
1023 for (auto* module : drive->GetConfig().GetModules()) {
1024 module->GetAzimuthMotorController()->SetKp(dt.Get());
1025 }
1026 break;
1027 case DoubleTelemetryField::ModulesAzimuthI:
1028 for (auto* module : drive->GetConfig().GetModules()) {
1029 module->GetAzimuthMotorController()->SetKi(dt.Get());
1030 }
1031 break;
1032 case DoubleTelemetryField::ModulesAzimuthD:
1033 for (auto* module : drive->GetConfig().GetModules()) {
1034 module->GetAzimuthMotorController()->SetKd(dt.Get());
1035 }
1036 break;
1037 case DoubleTelemetryField::ModulesDriveKs:
1038 for (auto* module : drive->GetConfig().GetModules()) {
1039 module->GetDriveMotorController()->SetKs(dt.Get());
1040 }
1041 break;
1042 case DoubleTelemetryField::ModulesDriveKv:
1043 for (auto* module : drive->GetConfig().GetModules()) {
1044 module->GetDriveMotorController()->SetKv(dt.Get());
1045 }
1046 break;
1047 case DoubleTelemetryField::ModulesDriveKa:
1048 for (auto* module : drive->GetConfig().GetModules()) {
1049 module->GetDriveMotorController()->SetKa(dt.Get());
1050 }
1051 break;
1052 case DoubleTelemetryField::ModulesAzimuthKs:
1053 for (auto* module : drive->GetConfig().GetModules()) {
1054 module->GetAzimuthMotorController()->SetKs(dt.Get());
1055 }
1056 break;
1057 case DoubleTelemetryField::ModulesAzimuthKv:
1058 for (auto* module : drive->GetConfig().GetModules()) {
1059 module->GetAzimuthMotorController()->SetKv(dt.Get());
1060 }
1061 break;
1062 case DoubleTelemetryField::ModulesAzimuthKa:
1063 for (auto* module : drive->GetConfig().GetModules()) {
1064 module->GetAzimuthMotorController()->SetKa(dt.Get());
1065 }
1066 break;
1067 default:
1068 break;
1069 }
1070 }
