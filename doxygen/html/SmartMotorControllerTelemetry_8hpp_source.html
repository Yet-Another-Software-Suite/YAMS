144template <typename F>
145class DoubleTelemetry {
146 public:
156 DoubleTelemetry(std::string key, double defaultVal, F field, bool tunable, std::string unit)
157 : m_field{field},
158 m_key{std::move(key)},
159 m_tunable{tunable},
160 m_unit{std::move(unit)},
161 m_defaultValue{defaultVal},
162 m_cachedValue{defaultVal} {}
163
169 void SetDefaultValue(double value) { m_cachedValue = m_defaultValue = value; }
170
177 void SetupNetworkTables(std::shared_ptr<wpi::nt::NetworkTable> dataTable,
178 std::shared_ptr<wpi::nt::NetworkTable> tuningTable) {
179 m_dataTable = dataTable;
180 m_tuningTable = tuningTable;
181 if (!m_enabled) return;
182
183 if (tuningTable && m_tunable) {
184 auto topic = tuningTable->GetDoubleTopic(m_key);
185 if (m_unit != "none") {
186 topic.SetProperties(wpi::util::json{{"units", m_unit}});
187 }
188 m_subPublisher = topic.Publish();
189 m_subPublisher->SetDefault(m_defaultValue);
190 m_subscriber = topic.Subscribe(m_defaultValue);
191 } else if (dataTable) {
192 auto topic = dataTable->GetDoubleTopic(m_key);
193 if (m_unit != "none") {
194 topic.SetProperties(wpi::util::json{{"units", m_unit}});
195 }
196 m_publisher = topic.Publish();
197 m_publisher->SetDefault(m_defaultValue);
198 }
199 }
200
206 void SetupNetworkTable(std::shared_ptr<wpi::nt::NetworkTable> dataTable) {
207 SetupNetworkTables(dataTable, nullptr);
208 }
209
215 void SetupDataLog(const std::string& prefix) {
216 if (m_tunable) return;
217 std::string path = prefix;
218 if (!path.empty() && path.back() != '/') path += '/';
219 path += m_unit + '/' + m_key;
220 m_dataLogEntry = wpi::log::DoubleLogEntry{wpi::DataLogManager::GetLog(), path};
221 }
222
229 void TransformUnit(const motorcontrollers::SmartMotorControllerConfig& cfg) {
230 bool linear = cfg.GetLinearClosedLoopControllerUse();
231 if (m_unit == "tunable_position") {
232 m_unit = linear ? "meter" : "degrees";
233 } else if (m_unit == "position") {
234 m_unit = linear ? "meter" : "rotations";
235 } else if (m_unit == "tunable_velocity") {
236 m_unit = linear ? "meter_per_second" : "rotations_per_minute";
237 } else if (m_unit == "velocity") {
238 m_unit = linear ? "meter_per_second" : "rotation_per_second";
239 } else if (m_unit == "tunable_acceleration") {
240 m_unit = linear ? "meter_per_second_per_second" : "rotations_per_minute_per_second";
241 } else if (m_unit == "acceleration") {
242 m_unit = linear ? "meter_per_second_per_second" : "rotation_per_second_per_second";
243 }
244 }
245
253 void ForceSet(double value) {
254 if (m_subPublisher) {
255 m_subPublisher->Set(value);
256 } else if (m_publisher) {
257 m_publisher->Set(value);
258 }
259 }
260
267 bool Set(double value) {
268 if (!m_enabled) return false;
269 if (m_dataLogEntry) {
270 m_dataLogEntry->Append(value);
271 }
272 if (m_subscriber) {
273 double tuningValue = m_subscriber->Get(m_defaultValue);
274 if (tuningValue != value) return false;
275 }
276 if (m_publisher) {
277 m_publisher->Set(value);
278 }
279 return true;
280 }
281
287 double Get() const {
288 if (!m_enabled) return m_defaultValue;
289 if (m_subscriber) {
290 return m_subscriber->Get(m_defaultValue);
291 }
292 throw std::runtime_error("Tuning table not configured for " + m_key + "!");
293 }
294
300 bool IsTunable() {
301 if (m_subscriber && m_tunable && m_enabled) {
302 double current = m_subscriber->Get(m_defaultValue);
303 if (current != m_cachedValue) {
304 m_cachedValue = current;
305 return true;
306 }
307 }
308 return false;
309 }
310
312 void Enable() { m_enabled = true; }
314 void Disable() { m_enabled = false; }
320 void Display(bool state) { m_enabled = state; }
322 F GetField() const { return m_field; }
324 bool IsEnabled() const { return m_enabled; }
325
327 void Close() {
328 m_subscriber.reset();
329 m_subPublisher.reset();
330 m_publisher.reset();
331 if (m_dataTable) {
332 m_dataTable->GetEntry(m_key).Unpublish();
333 }
334 if (m_tuningTable) {
335 m_tuningTable->GetEntry(m_key).Unpublish();
336 }
337 }
338
339 private:
340 F m_field;
341 std::string m_key;
342 bool m_tunable;
343 bool m_enabled{false};
344 std::string m_unit;
345 double m_defaultValue;
346 double m_cachedValue;
347
348 std::optional<wpi::nt::DoublePublisher> m_publisher;
349 std::optional<wpi::nt::DoubleSubscriber> m_subscriber;
350 std::optional<wpi::nt::DoublePublisher> m_subPublisher; // tunable: publish + subscribe
351 std::optional<wpi::log::DoubleLogEntry> m_dataLogEntry;
352
353 std::shared_ptr<wpi::nt::NetworkTable> m_tuningTable;
354 std::shared_ptr<wpi::nt::NetworkTable> m_dataTable;
355};
