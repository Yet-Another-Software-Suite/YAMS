121template <typename F>
122class DoubleTelemetry {
123 public:
133 DoubleTelemetry(std::string key, double defaultVal, F field, bool tunable, std::string unit)
134 : m_field{field},
135 m_key{std::move(key)},
136 m_tunable{tunable},
137 m_unit{std::move(unit)},
138 m_defaultValue{defaultVal},
139 m_cachedValue{defaultVal} {}
140
146 void SetDefaultValue(double value) { m_cachedValue = m_defaultValue = value; }
147
154 void SetupNetworkTables(std::shared_ptr<nt::NetworkTable> dataTable,
155 std::shared_ptr<nt::NetworkTable> tuningTable) {
156 m_dataTable = dataTable;
157 m_tuningTable = tuningTable;
158 if (!m_enabled) return;
159
160 if (tuningTable && m_tunable) {
161 auto topic = tuningTable->GetDoubleTopic(m_key);
162 if (m_unit != "none") {
163 topic.SetProperties(wpi::json{{"units", m_unit}});
164 }
165 m_subPublisher = topic.Publish();
166 m_subPublisher->SetDefault(m_defaultValue);
167 m_subscriber = topic.Subscribe(m_defaultValue);
168 } else if (dataTable) {
169 auto topic = dataTable->GetDoubleTopic(m_key);
170 if (m_unit != "none") {
171 topic.SetProperties(wpi::json{{"units", m_unit}});
172 }
173 m_publisher = topic.Publish();
174 m_publisher->SetDefault(m_defaultValue);
175 }
176 }
177
183 void SetupNetworkTable(std::shared_ptr<nt::NetworkTable> dataTable) {
184 SetupNetworkTables(dataTable, nullptr);
185 }
186
192 void SetupDataLog(const std::string& prefix) {
193 if (m_tunable) return;
194 std::string path = prefix;
195 if (!path.empty() && path.back() != '/') path += '/';
196 path += m_unit + '/' + m_key;
197 m_dataLogEntry = wpi::log::DoubleLogEntry{frc::DataLogManager::GetLog(), path};
198 }
199
206 void TransformUnit(const motorcontrollers::SmartMotorControllerConfig& cfg) {
207 bool linear = cfg.GetLinearClosedLoopControllerUse();
208 if (m_unit == "tunable_position") {
209 m_unit = linear ? "meter" : "degrees";
210 } else if (m_unit == "position") {
211 m_unit = linear ? "meter" : "rotations";
212 } else if (m_unit == "tunable_velocity") {
213 m_unit = linear ? "meter_per_second" : "rotations_per_minute";
214 } else if (m_unit == "velocity") {
215 m_unit = linear ? "meter_per_second" : "rotation_per_second";
216 } else if (m_unit == "tunable_acceleration") {
217 m_unit = linear ? "meter_per_second_per_second" : "rotations_per_minute_per_second";
218 } else if (m_unit == "acceleration") {
219 m_unit = linear ? "meter_per_second_per_second" : "rotation_per_second_per_second";
220 }
221 }
222
229 bool Set(double value) {
230 if (!m_enabled) return false;
231 if (m_dataLogEntry) {
232 m_dataLogEntry->Append(value);
233 }
234 if (m_subscriber) {
235 double tuningValue = m_subscriber->Get(m_defaultValue);
236 if (tuningValue != value) return false;
237 }
238 if (m_publisher) {
239 m_publisher->Set(value);
240 }
241 return true;
242 }
243
249 double Get() const {
250 if (!m_enabled) return m_defaultValue;
251 if (m_subscriber) {
252 return m_subscriber->Get(m_defaultValue);
253 }
254 throw std::runtime_error("Tuning table not configured for " + m_key + "!");
255 }
256
262 bool IsTunable() {
263 if (m_subscriber && m_tunable && m_enabled) {
264 double current = m_subscriber->Get(m_defaultValue);
265 if (current != m_cachedValue) {
266 m_cachedValue = current;
267 return true;
268 }
269 }
270 return false;
271 }
272
274 void Enable() { m_enabled = true; }
276 void Disable() { m_enabled = false; }
282 void Display(bool state) { m_enabled = state; }
284 F GetField() const { return m_field; }
286 bool IsEnabled() const { return m_enabled; }
287
289 void Close() {
290 m_subscriber.reset();
291 m_subPublisher.reset();
292 m_publisher.reset();
293 if (m_dataTable) {
294 m_dataTable->GetEntry(m_key).Unpublish();
295 }
296 if (m_tuningTable) {
297 m_tuningTable->GetEntry(m_key).Unpublish();
298 }
299 }
300
301 private:
302 F m_field;
303 std::string m_key;
304 bool m_tunable;
305 bool m_enabled{false};
306 std::string m_unit;
307 double m_defaultValue;
308 double m_cachedValue;
309
310 std::optional<nt::DoublePublisher> m_publisher;
311 std::optional<nt::DoubleSubscriber> m_subscriber;
312 std::optional<nt::DoublePublisher> m_subPublisher; // tunable: publish + subscribe
313 std::optional<wpi::log::DoubleLogEntry> m_dataLogEntry;
314
315 std::shared_ptr<nt::NetworkTable> m_tuningTable;
316 std::shared_ptr<nt::NetworkTable> m_dataTable;
317};
