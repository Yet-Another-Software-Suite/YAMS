127template <typename F>
128class DoubleTelemetry {
129 public:
139 DoubleTelemetry(std::string key, double defaultVal, F field, bool tunable, std::string unit)
140 : m_field{field},
141 m_key{std::move(key)},
142 m_tunable{tunable},
143 m_unit{std::move(unit)},
144 m_defaultValue{defaultVal},
145 m_cachedValue{defaultVal} {}
146
152 void SetDefaultValue(double value) { m_cachedValue = m_defaultValue = value; }
153
160 void SetupNetworkTables(std::shared_ptr<wpi::nt::NetworkTable> dataTable,
161 std::shared_ptr<wpi::nt::NetworkTable> tuningTable) {
162 m_dataTable = dataTable;
163 m_tuningTable = tuningTable;
164 if (!m_enabled) return;
165
166 if (tuningTable && m_tunable) {
167 auto topic = tuningTable->GetDoubleTopic(m_key);
168 if (m_unit != "none") {
169 topic.SetProperties(wpi::util::json{{"units", m_unit}});
170 }
171 m_subPublisher = topic.Publish();
172 m_subPublisher->SetDefault(m_defaultValue);
173 m_subscriber = topic.Subscribe(m_defaultValue);
174 } else if (dataTable) {
175 auto topic = dataTable->GetDoubleTopic(m_key);
176 if (m_unit != "none") {
177 topic.SetProperties(wpi::util::json{{"units", m_unit}});
178 }
179 m_publisher = topic.Publish();
180 m_publisher->SetDefault(m_defaultValue);
181 }
182 }
183
189 void SetupNetworkTable(std::shared_ptr<wpi::nt::NetworkTable> dataTable) {
190 SetupNetworkTables(dataTable, nullptr);
191 }
192
198 void SetupDataLog(const std::string& prefix) {
199 if (m_tunable) return;
200 std::string path = prefix;
201 if (!path.empty() && path.back() != '/') path += '/';
202 path += m_unit + '/' + m_key;
203 m_dataLogEntry = wpi::log::DoubleLogEntry{wpi::DataLogManager::GetLog(), path};
204 }
205
212 void TransformUnit(const motorcontrollers::SmartMotorControllerConfig& cfg) {
213 bool linear = cfg.GetLinearClosedLoopControllerUse();
214 if (m_unit == "tunable_position") {
215 m_unit = linear ? "meter" : "degrees";
216 } else if (m_unit == "position") {
217 m_unit = linear ? "meter" : "rotations";
218 } else if (m_unit == "tunable_velocity") {
219 m_unit = linear ? "meter_per_second" : "rotations_per_minute";
220 } else if (m_unit == "velocity") {
221 m_unit = linear ? "meter_per_second" : "rotation_per_second";
222 } else if (m_unit == "tunable_acceleration") {
223 m_unit = linear ? "meter_per_second_per_second" : "rotations_per_minute_per_second";
224 } else if (m_unit == "acceleration") {
225 m_unit = linear ? "meter_per_second_per_second" : "rotation_per_second_per_second";
226 }
227 }
228
236 void ForceSet(double value) {
237 if (m_subPublisher) {
238 m_subPublisher->Set(value);
239 } else if (m_publisher) {
240 m_publisher->Set(value);
241 }
242 }
243
250 bool Set(double value) {
251 if (!m_enabled) return false;
252 if (m_dataLogEntry) {
253 m_dataLogEntry->Append(value);
254 }
255 if (m_subscriber) {
256 double tuningValue = m_subscriber->Get(m_defaultValue);
257 if (tuningValue != value) return false;
258 }
259 if (m_publisher) {
260 m_publisher->Set(value);
261 }
262 return true;
263 }
264
270 double Get() const {
271 if (!m_enabled) return m_defaultValue;
272 if (m_subscriber) {
273 return m_subscriber->Get(m_defaultValue);
274 }
275 throw std::runtime_error("Tuning table not configured for " + m_key + "!");
276 }
277
283 bool IsTunable() {
284 if (m_subscriber && m_tunable && m_enabled) {
285 double current = m_subscriber->Get(m_defaultValue);
286 if (current != m_cachedValue) {
287 m_cachedValue = current;
288 return true;
289 }
290 }
291 return false;
292 }
293
295 void Enable() { m_enabled = true; }
297 void Disable() { m_enabled = false; }
303 void Display(bool state) { m_enabled = state; }
305 F GetField() const { return m_field; }
307 bool IsEnabled() const { return m_enabled; }
308
310 void Close() {
311 m_subscriber.reset();
312 m_subPublisher.reset();
313 m_publisher.reset();
314 if (m_dataTable) {
315 m_dataTable->GetEntry(m_key).Unpublish();
316 }
317 if (m_tuningTable) {
318 m_tuningTable->GetEntry(m_key).Unpublish();
319 }
320 }
321
322 private:
323 F m_field;
324 std::string m_key;
325 bool m_tunable;
326 bool m_enabled{false};
327 std::string m_unit;
328 double m_defaultValue;
329 double m_cachedValue;
330
331 std::optional<wpi::nt::DoublePublisher> m_publisher;
332 std::optional<wpi::nt::DoubleSubscriber> m_subscriber;
333 std::optional<wpi::nt::DoublePublisher> m_subPublisher; // tunable: publish + subscribe
334 std::optional<wpi::log::DoubleLogEntry> m_dataLogEntry;
335
336 std::shared_ptr<wpi::nt::NetworkTable> m_tuningTable;
337 std::shared_ptr<wpi::nt::NetworkTable> m_dataTable;
338};
