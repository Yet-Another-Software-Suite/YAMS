130 wpi::math::Pose3d GetTranslatedPose(wpi::units::meter_t distance, wpi::units::radian_t angle) const {
131 const wpi::units::meter_t along = distance * std::cos(angle.value());
132 const wpi::units::meter_t across = distance * std::sin(angle.value());
133 wpi::math::Translation3d slide;
134 switch (m_plane) {
135 case Plane::XZ:
136 slide = wpi::math::Translation3d{along, wpi::units::meter_t{0}, across};
137 break;
138 case Plane::YZ:
139 slide = wpi::math::Translation3d{wpi::units::meter_t{0}, along, across};
140 break;
141 case Plane::XY:
142 slide = wpi::math::Translation3d{along, across, wpi::units::meter_t{0}};
143 break;
144 }
145 return GetMountPose().TransformBy(wpi::math::Transform3d{slide, wpi::math::Rotation3d{}});
146 }

154 wpi::math::Rotation3d RotationInPlane(wpi::units::radian_t angle) const {
155 switch (m_plane) {
156 case Plane::YZ:
157 // +Y turns toward +Z: a positive roll about +X.
158 return wpi::math::Rotation3d{angle, wpi::units::radian_t{0}, wpi::units::radian_t{0}};
159 case Plane::XY:
160 // +X turns toward +Y: a positive yaw about +Z.
161 return wpi::math::Rotation3d{wpi::units::radian_t{0}, wpi::units::radian_t{0}, angle};
162 case Plane::XZ:
163 default:
164 // +X turns toward +Z: a negative pitch about +Y.
165 return wpi::math::Rotation3d{wpi::units::radian_t{0}, -angle, wpi::units::radian_t{0}};
166 }
