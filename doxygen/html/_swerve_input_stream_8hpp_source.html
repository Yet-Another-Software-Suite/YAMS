365 wpi::math::ChassisVelocities Get() {
366 auto& cfg = m_swerveDrive->GetConfig();
367
368 double maxLinear =
369 cfg.GetMaximumChassisLinearVelocity().value_or(m_maximumChassisLinearVelocity).value();
370 wpi::units::radians_per_second_t maxAngular =
371 cfg.GetMaximumChassisAngularVelocity().has_value()
372 ? wpi::units::radians_per_second_t{cfg.GetMaximumChassisAngularVelocity().value()}
373 : m_maximumChassisAngularVelocity;
374
375 wpi::math::Translation2d scaledTranslation = ApplyTranslationScalar(
376 ApplyDeadband(m_controllerTranslationX()), ApplyDeadband(m_controllerTranslationY()));
377 scaledTranslation = ApplyTranslationCube(scaledTranslation);
378 scaledTranslation = ApplyAllianceAwareTranslation(scaledTranslation);
379
380 double vx = scaledTranslation.X().value() * maxLinear;
381 double vy = scaledTranslation.Y().value() * maxLinear;
382 double omega = 0.0;
383 wpi::math::ChassisVelocities speeds{};
384
385 SwerveInputMode newMode = FindMode();
386 if (m_currentMode != newMode) {
387 TransitionMode(newMode);
388 }
389
390 switch (newMode) {
391 case SwerveInputMode::TRANSLATION_ONLY: {
392 auto& pid = RequireRotationPID();
393 omega = pid.Calculate(wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}.value(),
394 m_lockedHeading.value().Radians().value());
395 break;
396 }
397 case SwerveInputMode::ANGULAR_VELOCITY: {
398 omega = ApplyOmegaCube(ApplyRotationalScalar(ApplyDeadband(m_controllerOmega.value()()))) *
399 maxAngular.value();
400 break;
401 }
402 case SwerveInputMode::HEADING: {
403 auto& pid = RequireRotationPID();
404 omega = pid.Calculate(wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}.value(),
405 m_headingSupplier.value()().value());
406 // Prevent rotation if controller heading inputs are not past axisDeadband
407 if (m_controllerHeadingX.has_value() && m_controllerHeadingY.has_value() &&
408 m_axisDeadband.has_value() &&
409 std::abs(m_controllerHeadingX.value()()) + std::abs(m_controllerHeadingY.value()()) <
410 m_axisDeadband.value()) {
411 omega = 0.0;
412 }
413 break;
414 }
415 case SwerveInputMode::AIM: {
416 auto& pid = RequireRotationPID();
417 auto currentHeading =
418 wpi::math::Rotation2d{wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}};
419 auto relativeTrl = m_aimTarget.value()().RelativeTo(m_swerveDrive->GetPose()).Translation();
420 auto target = relativeTrl.Angle().value_or(wpi::math::Rotation2d{}) + currentHeading;
421 omega = pid.Calculate(currentHeading.Radians().value(), target.Radians().value());
422 break;
423 }
424 }
425
426 m_currentMode = newMode;
427 speeds = wpi::math::ChassisVelocities{wpi::units::meters_per_second_t{vx},
428 wpi::units::meters_per_second_t{vy},
429 wpi::units::radians_per_second_t{omega}};
430 return ApplyTranslationHeadingOffset(ApplyRobotRelativeTranslation(speeds));
431 }

495 wpi::math::PIDController& RequireRotationPID() {
496 auto pid = m_swerveDrive->GetConfig().GetRotationPID();
497 if (!pid) {
498 throw exceptions::SwerveDriveConfigurationException(
499 "No rotation PID controller configured, heading, aim, and translation only control are "
500 "unavailable. Use SwerveDriveConfig::WithRotationController(PIDController) to fix this "
501 "error.");
502 }
503 return pid->get();
504 }

506 SwerveInputMode FindMode() {
507 if (m_translationOnlyEnabled.has_value() && m_translationOnlyEnabled.value()()) {
508 return SwerveInputMode::TRANSLATION_ONLY;
509 }
510 if (m_aimEnabled.has_value() && m_aimEnabled.value()()) {
511 if (m_aimTarget.has_value()) {
512 return SwerveInputMode::AIM;
513 }
514 std::cerr << "[YAMS SwerveInputStream] AIM mode enabled but no target set. "
515 "Call WithAim() first.\n";
516 }
517 if (m_headingEnabled.has_value() && m_headingEnabled.value()()) {
518 if (m_headingSupplier.has_value()) {
519 return SwerveInputMode::HEADING;
520 }
521 std::cerr << "[YAMS SwerveInputStream] HEADING mode enabled but no heading set. "
522 "Call WithHeading() or WithControllerHeadingAxis() first.\n";
523 }
524 if (!m_controllerOmega.has_value()) {
525 std::cerr << "[YAMS SwerveInputStream] No rotation axis configured. "
526 "Call WithControllerRotationAxis() or WithControllerHeadingAxis(). "
527 "Falling back to TRANSLATION_ONLY.\n";
528 return SwerveInputMode::TRANSLATION_ONLY;
529 }
530 return SwerveInputMode::ANGULAR_VELOCITY;
531 }

533 void TransitionMode(SwerveInputMode newMode) {
534 // Exit current mode
535 switch (m_currentMode) {
536 case SwerveInputMode::TRANSLATION_ONLY:
537 m_lockedHeading = std::nullopt;
538 break;
539 case SwerveInputMode::HEADING:
540 case SwerveInputMode::AIM:
541 m_swerveDrive->ResetRotationPID();
542 break;
543 default:
544 break;
545 }
546 // Enter new mode
547 switch (newMode) {
548 case SwerveInputMode::TRANSLATION_ONLY:
549 m_lockedHeading =
550 wpi::math::Rotation2d{wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}};
551 m_swerveDrive->ResetRotationPID();
552 break;
553 case SwerveInputMode::HEADING:
554 case SwerveInputMode::AIM:
555 m_swerveDrive->ResetRotationPID();
556 break;
557 default:
558 break;
559 }
560 }

600 wpi::math::Translation2d ApplyAllianceAwareTranslation(wpi::math::Translation2d translation) {
601 // Robot relative translation does not depend on the alliance, so it is never flipped.
602 if (m_robotRelative.has_value() && m_robotRelative.value()()) {
603 return translation;
604 }
605 if (m_allianceRelative.has_value() && m_allianceRelative.value()()) {
606 auto alliance = wpi::MatchState::GetAlliance();
607 if (alliance.has_value() && alliance.value() == wpi::Alliance::RED) {
608 return translation.RotateBy(wpi::math::Rotation2d{wpi::units::degree_t{180.0}});
609 }
610 }
611 return translation;
612 }
613
614 wpi::math::ChassisVelocities ApplyTranslationHeadingOffset(wpi::math::ChassisVelocities speeds) {
615 if (m_translationHeadingOffsetEnabled.has_value() &&
616 m_translationHeadingOffsetEnabled.value()() && m_translationHeadingOffset.has_value()) {
617 wpi::math::Translation2d vec{wpi::units::meter_t{speeds.vx.value()},
618 wpi::units::meter_t{speeds.vy.value()}};
619 auto rotated = vec.RotateBy(m_translationHeadingOffset.value());
620 return wpi::math::ChassisVelocities{wpi::units::meters_per_second_t{rotated.X().value()},
621 wpi::units::meters_per_second_t{rotated.Y().value()},
622 speeds.omega};
623 }
624 return speeds;
