286 wpi::math::ChassisVelocities Get() {
287 // Starts from the drive config's maximums (see the constructor); WithMaximum*Velocity and live
288 // tuning override them.
289 double maxLinear = m_maximumChassisLinearVelocity.value();
290 wpi::units::radians_per_second_t maxAngular = m_maximumChassisAngularVelocity;
291
292 wpi::math::Translation2d scaledTranslation = ApplyTranslationScalar(
293 ApplyDeadband(m_controllerTranslationX()), ApplyDeadband(m_controllerTranslationY()));
294 scaledTranslation = ApplyTranslationCube(scaledTranslation);
295 scaledTranslation = ApplyAllianceAwareTranslation(scaledTranslation);
296
297 double vx = scaledTranslation.X().value() * maxLinear;
298 double vy = scaledTranslation.Y().value() * maxLinear;
299 double omega = 0.0;
300 wpi::math::ChassisVelocities speeds{};
301
302 SwerveInputMode newMode = FindMode();
303 if (m_currentMode != newMode) {
304 TransitionMode(newMode);
305 }
306
307 switch (newMode) {
308 case SwerveInputMode::TRANSLATION_ONLY: {
309 auto& pid = RequireRotationPID();
310 omega = pid.Calculate(wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}.value(),
311 m_lockedHeading.value().Radians().value());
312 break;
313 }
314 case SwerveInputMode::ANGULAR_VELOCITY: {
315 omega = ApplyOmegaCube(ApplyRotationalScalar(ApplyDeadband(m_controllerOmega.value()()))) *
316 maxAngular.value();
317 break;
318 }
319 case SwerveInputMode::HEADING: {
320 auto& pid = RequireRotationPID();
321 omega = pid.Calculate(wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}.value(),
322 m_headingSupplier.value()().value());
323 if (m_controllerHeadingX.has_value() && m_controllerHeadingY.has_value() &&
324 m_axisDeadband.has_value() &&
325 std::abs(m_controllerHeadingX.value()()) + std::abs(m_controllerHeadingY.value()()) <
326 m_axisDeadband.value()) {
327 omega = 0.0;
328 }
329 break;
330 }
331 case SwerveInputMode::AIM: {
332 auto& pid = RequireRotationPID();
333 auto currentHeading =
334 wpi::math::Rotation2d{wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}};
335 auto relativeTrl = m_aimTarget.value()().RelativeTo(m_swerveDrive->GetPose()).Translation();
336 auto target = relativeTrl.Angle().value_or(wpi::math::Rotation2d{}) + currentHeading;
337 omega = pid.Calculate(currentHeading.Radians().value(), target.Radians().value());
338 break;
339 }
340 }
341
342 m_currentMode = newMode;
343 speeds = wpi::math::ChassisVelocities{wpi::units::meters_per_second_t{vx},
344 wpi::units::meters_per_second_t{vy},
345 wpi::units::radians_per_second_t{omega}};
346 auto fieldRelativeSpeeds = ApplyTranslationHeadingOffset(ApplyRobotRelativeTranslation(speeds));
347 if (auto telemetry = GetTelemetry()) {
348 telemetry->get().UpdateTelemetry();
349 }
350 return fieldRelativeSpeeds;
351 }

447 SwerveInputMode FindMode() {
448 if (m_translationOnlyEnabled.has_value() && m_translationOnlyEnabled.value()()) {
449 return SwerveInputMode::TRANSLATION_ONLY;
450 }
451 if (m_aimEnabled.has_value() && m_aimEnabled.value()()) {
452 if (m_aimTarget.has_value()) {
453 return SwerveInputMode::AIM;
454 }
455 std::cerr << "[YAMS SwerveInputStream] AIM mode enabled but no target set. Call WithAim() first.\n";
456 }
457 if (m_headingEnabled.has_value() && m_headingEnabled.value()()) {
458 if (m_headingSupplier.has_value()) {
459 return SwerveInputMode::HEADING;
460 }
461 std::cerr << "[YAMS SwerveInputStream] HEADING mode enabled but no heading set. "
462 "Call WithHeading() or WithControllerHeadingAxis() first.\n";
463 }
464 if (!m_controllerOmega.has_value()) {
465 std::cerr << "[YAMS SwerveInputStream] No rotation axis configured. "
466 "Call WithControllerRotationAxis() or WithControllerHeadingAxis(). "
467 "Falling back to TRANSLATION_ONLY.\n";
468 return SwerveInputMode::TRANSLATION_ONLY;
469 }
470 return SwerveInputMode::ANGULAR_VELOCITY;
471 }

473 void TransitionMode(SwerveInputMode newMode) {
474 switch (m_currentMode) {
475 case SwerveInputMode::TRANSLATION_ONLY:
476 m_lockedHeading = std::nullopt;
477 break;
478 case SwerveInputMode::HEADING:
479 case SwerveInputMode::AIM:
480 m_swerveDrive->ResetRotationPID();
481 break;
482 default:
483 break;
484 }
485 switch (newMode) {
486 case SwerveInputMode::TRANSLATION_ONLY:
487 m_lockedHeading =
488 wpi::math::Rotation2d{wpi::units::radian_t{m_swerveDrive->GetGyroAngle()}};
489 m_swerveDrive->ResetRotationPID();
490 break;
491 case SwerveInputMode::HEADING:
492 case SwerveInputMode::AIM:
493 m_swerveDrive->ResetRotationPID();
494 break;
495 default:
496 break;
497 }
498 }

538 wpi::math::Translation2d ApplyAllianceAwareTranslation(wpi::math::Translation2d translation) {
539 if (m_robotRelative.has_value() && m_robotRelative.value()()) {
540 return translation;
541 }
542 if (m_allianceRelative.has_value() && m_allianceRelative.value()()) {
543 auto alliance = wpi::MatchState::GetAlliance();
544 if (alliance.has_value() && alliance.value() == wpi::Alliance::RED) {
545 return translation.RotateBy(wpi::math::Rotation2d{wpi::units::degree_t{180.0}});
546 }
547 }
548 return translation;
549 }
550
551 wpi::math::ChassisVelocities ApplyTranslationHeadingOffset(wpi::math::ChassisVelocities speeds) {
552 if (m_translationHeadingOffsetEnabled.has_value() &&
553 m_translationHeadingOffsetEnabled.value()() && m_translationHeadingOffset.has_value()) {
554 wpi::math::Translation2d vec{wpi::units::meter_t{speeds.vx.value()},
555 wpi::units::meter_t{speeds.vy.value()}};
556 auto rotated = vec.RotateBy(m_translationHeadingOffset.value());
557 return wpi::math::ChassisVelocities{wpi::units::meters_per_second_t{rotated.X().value()},
558 wpi::units::meters_per_second_t{rotated.Y().value()},
559 speeds.omega};
560 }
561 return speeds;
562 }
