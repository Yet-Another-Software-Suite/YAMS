514 frc::ChassisSpeeds Get() {
515 double maxLinear = m_maximumChassisLinearVelocity.value();
516 units::radians_per_second_t maxAngular = m_maximumChassisAngularVelocity;
517
518 frc::Translation2d scaledTranslation = ApplyTranslationScalar(
519 ApplyDeadband(m_controllerTranslationX()), ApplyDeadband(m_controllerTranslationY()));
520 scaledTranslation = ApplyTranslationCube(scaledTranslation);
521 scaledTranslation = ApplyAllianceAwareTranslation(scaledTranslation);
522
523 double vx = scaledTranslation.X().value() * maxLinear;
524 double vy = scaledTranslation.Y().value() * maxLinear;
525 double omega = 0.0;
526 frc::ChassisSpeeds speeds{};
527
528 SwerveInputMode newMode = FindMode();
529 if (m_currentMode != newMode) {
530 TransitionMode(newMode);
531 }
532
533 switch (newMode) {
534 case SwerveInputMode::TRANSLATION_ONLY: {
535 auto& pid = RequireRotationPID();
536 omega = pid.Calculate(units::radian_t{m_swerveDrive->GetGyroAngle()}.value(),
537 m_lockedHeading.value().Radians().value());
538 break;
539 }
540 case SwerveInputMode::ANGULAR_VELOCITY: {
541 omega = ApplyOmegaCube(ApplyRotationalScalar(ApplyDeadband(m_controllerOmega.value()()))) *
542 maxAngular.value();
543 break;
544 }
545 case SwerveInputMode::HEADING: {
546 auto& pid = RequireRotationPID();
547 omega = pid.Calculate(units::radian_t{m_swerveDrive->GetGyroAngle()}.value(),
548 m_headingSupplier.value()().value());
549 // Do not turn while the heading stick is inside the deadband.
550 if (m_controllerHeadingX.has_value() && m_controllerHeadingY.has_value() &&
551 m_axisDeadband.has_value() &&
552 std::abs(m_controllerHeadingX.value()()) + std::abs(m_controllerHeadingY.value()()) <
553 m_axisDeadband.value()) {
554 omega = 0.0;
555 }
556 break;
557 }
558 case SwerveInputMode::AIM: {
559 auto& pid = RequireRotationPID();
560 auto currentHeading = frc::Rotation2d{units::radian_t{m_swerveDrive->GetGyroAngle()}};
561 auto relativeTrl = m_aimTarget.value()().RelativeTo(m_swerveDrive->GetPose()).Translation();
562 auto target = relativeTrl.Angle() + currentHeading;
563 omega = pid.Calculate(currentHeading.Radians().value(), target.Radians().value());
564 break;
565 }
566 }
567
568 m_currentMode = newMode;
569 speeds = frc::ChassisSpeeds{units::meters_per_second_t{vx}, units::meters_per_second_t{vy},
570 units::radians_per_second_t{omega}};
571 auto fieldRelativeSpeeds = ApplyTranslationHeadingOffset(ApplyRobotRelativeTranslation(speeds));
572 if (auto telemetry = GetTelemetry()) {
573 telemetry->get().UpdateTelemetry();
574 }
575 return fieldRelativeSpeeds;
576 }

688 SwerveInputMode FindMode() {
689 if (m_translationOnlyEnabled.has_value() && m_translationOnlyEnabled.value()()) {
690 return SwerveInputMode::TRANSLATION_ONLY;
691 }
692 if (m_aimEnabled.has_value() && m_aimEnabled.value()()) {
693 if (m_aimTarget.has_value()) {
694 return SwerveInputMode::AIM;
695 }
696 std::cerr << "[YAMS SwerveInputStream] AIM mode enabled but no target set. "
697 "Call WithAim() first.\n";
698 }
699 if (m_headingEnabled.has_value() && m_headingEnabled.value()()) {
700 if (m_headingSupplier.has_value()) {
701 return SwerveInputMode::HEADING;
702 }
703 std::cerr << "[YAMS SwerveInputStream] HEADING mode enabled but no heading set. "
704 "Call WithHeading() or WithControllerHeadingAxis() first.\n";
705 }
706 if (!m_controllerOmega.has_value()) {
707 std::cerr << "[YAMS SwerveInputStream] No rotation axis configured. "
708 "Call WithControllerRotationAxis() or WithControllerHeadingAxis(). "
709 "Falling back to TRANSLATION_ONLY.\n";
710 return SwerveInputMode::TRANSLATION_ONLY;
711 }
712 return SwerveInputMode::ANGULAR_VELOCITY;
713 }

715 void TransitionMode(SwerveInputMode newMode) {
716 // Exit current mode
717 switch (m_currentMode) {
718 case SwerveInputMode::TRANSLATION_ONLY:
719 m_lockedHeading = std::nullopt;
720 break;
721 case SwerveInputMode::HEADING:
722 case SwerveInputMode::AIM:
723 m_swerveDrive->ResetRotationPID();
724 break;
725 default:
726 break;
727 }
728 // Enter new mode
729 switch (newMode) {
730 case SwerveInputMode::TRANSLATION_ONLY:
731 m_lockedHeading = frc::Rotation2d{units::radian_t{m_swerveDrive->GetGyroAngle()}};
732 m_swerveDrive->ResetRotationPID();
733 break;
734 case SwerveInputMode::HEADING:
735 case SwerveInputMode::AIM:
736 m_swerveDrive->ResetRotationPID();
737 break;
738 default:
739 break;
740 }
741 }

781 frc::Translation2d ApplyAllianceAwareTranslation(frc::Translation2d translation) {
782 // Robot relative translation is not on the field, so there is nothing to flip for the alliance.
783 if (m_robotRelative.has_value() && m_robotRelative.value()()) {
784 return translation;
785 }
786 if (m_allianceRelative.has_value() && m_allianceRelative.value()()) {
787 auto alliance = frc::DriverStation::GetAlliance();
788 if (alliance.has_value() && alliance.value() == frc::DriverStation::Alliance::kRed) {
789 return translation.RotateBy(frc::Rotation2d{units::degree_t{180.0}});
790 }
791 }
792 return translation;
793 }
794
795 frc::ChassisSpeeds ApplyTranslationHeadingOffset(frc::ChassisSpeeds speeds) {
796 if (m_translationHeadingOffsetEnabled.has_value() &&
797 m_translationHeadingOffsetEnabled.value()() && m_translationHeadingOffset.has_value()) {
798 frc::Translation2d vec{units::meter_t{speeds.vx.value()}, units::meter_t{speeds.vy.value()}};
799 auto rotated = vec.RotateBy(m_translationHeadingOffset.value());
800 return frc::ChassisSpeeds{units::meters_per_second_t{rotated.X().value()},
801 units::meters_per_second_t{rotated.Y().value()}, speeds.omega};
802 }
803 return speeds;
